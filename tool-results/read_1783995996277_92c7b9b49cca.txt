     1→     1→/**
     2→     2→ * Chat Store — Centralized state management for agent chat.
     3→     3→ *
     4→     4→ * Key features (the "1-up" philosophy):
     5→     5→ *  - **Message queue**: users can queue multiple messages while the LLM is
     6→     6→ *    working. They fire sequentially the moment the previous turn ends.
     7→     7→ *  - **SSE streaming**: real-time deltas (no 300ms polling pop-in).
     8→     8→ *  - **Crash-safe persistence**: activeSessionId and workspaceId survive
     9→     9→ *    refresh. The session list is rehydrated from the backend on mount.
    10→    10→ *  - **Optimistic UI**: every user action reflects instantly in the UI;
    11→    11→ *    backend errors are surfaced as inline retries, never as disappearing
    12→    12→ *    state.
    13→    13→ *  - **Immutable updates**: no in-place mutation of messages (which caused
    14→    14→ *    the previous "stack weirdly" bug).
    15→    15→ *  - **Single source of truth for the agent session**: we track
    16→    16→ *    `currentAgentSessionId` so Stop actually works.
    17→    17→ */

   162→   162→/** Convert raw AgentEvents into normalized ChatMessages. Pure function. */
   163→   163→export function eventsToMessages(events: AgentEvent[]): ChatMessage[] {
   164→   164→  const messages: ChatMessage[] = [];
   165→   165→  // Track the index of the streaming assistant / thinking message so deltas
   166→   166→  // merge into it without mutation.
   167→   167→  let streamingAssistantIdx = -1;
   168→   168→  let streamingThinkingIdx = -1;
   169→   169→
   170→   170→  for (const ev of events) {
   171→   171→    const ts = (ev.ts || Date.now() / 1000) * 1000;
   172→   172→    const seq = ev.i ?? messages.length;
   173→   173→
   174→   174→    switch (ev.type) {
   175→   175→      case "user": {
   176→   176→        messages.push({
   177→   177→          id: genMsgId(),
   178→   178→          role: "user",
   179→   179→          content: ev.text || "",
   180→   180→          timestamp: ts,
   181→   181→          seq,
   182→   182→        });
   183→   183→        streamingAssistantIdx = -1;
   184→   184→        streamingThinkingIdx = -1;
   185→   185→        break;
   186→   186→      }
   187→   187→      case "assistant": {
   188→   188→        messages.push({
   189→   189→          id: genMsgId(),
   190→   190→          role: "assistant",
   191→   191→          content: ev.text || "",
   192→   192→          timestamp: ts,
   193→   193→          seq,
   194→   194→        });
   195→   195→        streamingAssistantIdx = -1;
   196→   196→        streamingThinkingIdx = -1;
   197→   197→        break;
   198→   198→      }
   199→   199→      case "assistant_delta": {
   200→   200→        if (streamingAssistantIdx === -1) {
   201→   201→          messages.push({
   202→   202→            id: genMsgId(),
   203→   203→            role: "assistant",
   204→   204→            content: ev.text || "",
   205→   205→            isStreaming: true,
   206→   206→            timestamp: ts,
   207→   207→            seq,
   208→   208→          });
   209→   209→          streamingAssistantIdx = messages.length - 1;
   210→   210→        } else {
   211→   211→          const cur = messages[streamingAssistantIdx];
   212→   212→          messages[streamingAssistantIdx] = {
   213→   213→            ...cur,
   214→   214→            content: cur.content + (ev.text || ""),
   215→   215→            seq,
   216→   216→            isStreaming: true,
   217→   217→          };
   218→   218→        }
   219→   219→        streamingThinkingIdx = -1;
   220→   220→        break;
   221→   221→      }
   222→   222→      case "thinking": {
   223→   223→        messages.push({
   224→   224→          id: genMsgId(),
   225→   225→          role: "thinking",
   226→   226→          content: ev.text || "",
   227→   227→          timestamp: ts,
   228→   228→          seq,
   229→   229→        });
   230→   230→        streamingThinkingIdx = -1;
   231→   231→        streamingAssistantIdx = -1;
   232→   232→        break;
   233→   233→      }
   234→   234→      case "thinking_delta": {
   235→   235→        if (streamingThinkingIdx === -1) {
   236→   236→          messages.push({
   237→   237→            id: genMsgId(),
   238→   238→            role: "thinking",
   239→   239→            content: ev.text || "",
   240→   240→            isStreaming: true,
   241→   241→            timestamp: ts,
   242→   242→            seq,
   243→   243→          });
   244→   244→          streamingThinkingIdx = messages.length - 1;
   245→   245→        } else {
   246→   246→          const cur = messages[streamingThinkingIdx];
   247→   247→          messages[streamingThinkingIdx] = {
   248→   248→            ...cur,
   249→   249→            content: cur.content + (ev.text || ""),
   250→   250→            seq,
   251→   251→            isStreaming: true,
   252→   252→          };
   253→   253→        }
   254→   254→        streamingAssistantIdx = -1;
   255→   255→        break;
   256→   256→      }
   257→   257→      case "tool_use": {
   258→   258→        messages.push({
   259→   259→          id: genMsgId(),
   260→   260→          role: "tool",
   261→   261→          content: "",
   262→   262→          toolName: ev.name || "tool",
   263→   263→          toolSummary: ev.summary || "",
   264→   264→          timestamp: ts,
   265→   265→          seq,
   266→   266→        });
   267→   267→        streamingAssistantIdx = -1;
   268→   268→        streamingThinkingIdx = -1;
   269→   269→        break;
   270→   270→      }
   271→   271→      case "tool_result": {
   272→   272→        messages.push({
   273→   273→          id: genMsgId(),
   274→   274→          role: "tool",
   275→   275→          content: ev.text || "",
   276→   276→          isError: ev.is_error,
   277→   277→          toolName: "result",
   278→   278→          timestamp: ts,
   279→   279→          seq,
   280→   280→        });
   281→   281→        streamingAssistantIdx = -1;
   282→   282→        streamingThinkingIdx = -1;
   283→   283→        break;
   284→   284→      }
   285→   285→      case "status": {
   286→   286→        const st = ev as {
   287→   287→          state: AgentStatus;
   288→   288→          detail?: string;
   289→   289→          cost_usd?: number | null;
   290→   290→        };
   291→   291→        if (st.state === "idle" && st.detail === "interrupted") {
   292→   292→          messages.push({
   293→   293→            id: genMsgId(),
   294→   294→            role: "status",
   295→   295→            content: "— stopped —",
   296→   296→            timestamp: ts,
   297→   297→            seq,
   298→   298→          });
   299→   299→        } else if (st.state === "error") {
   300→   300→          messages.push({
   301→   301→            id: genMsgId(),
   302→   302→            role: "status",
   303→   303→            content: st.detail || "Agent error",
   304→   304→            isError: true,
   305→   305→            costUsd: st.cost_usd,
   306→   306→            timestamp: ts,
   307→   307→            seq,
   308→   308→          });
   309→   309→        }
   310→   310→        // status events don't reset streaming indexes — they wrap a turn.
   311→   311→        break;
   312→   312→      }
   313→   313→      case "panel": {
   314→   314→        messages.push({
   315→   315→          id: genMsgId(),
   316→   316→          role: "panel",
   317→   317→          content: ev.task_name || "Panel",
   318→   318→          panelStatus: ev.status || "starting",
   319→   319→          timestamp: ts,
   320→   320→          seq,
   321→   321→        });
   322→   322→        streamingAssistantIdx = -1;
   323→   323→        streamingThinkingIdx = -1;
   324→   324→        break;
   325→   325→      }
   326→   326→      // steering events are not rendered as messages (they're routing metadata).
   327→   327→    }
   328→   328→  }
   329→   329→
   330→   330→  // Mark the last streaming message as no longer streaming (turn ended).
   331→   331→  if (streamingAssistantIdx !== -1) {
   332→   332→    messages[streamingAssistantIdx] = { ...messages[streamingAssistantIdx], isStreaming: false };
   333→   333→  }
   334→   334→  if (streamingThinkingIdx !== -1) {
   335→   335→    messages[streamingThinkingIdx] = { ...messages[streamingThinkingIdx], isStreaming: false };
   336→   336→  }
   337→   337→
   338→   338→  return messages;
   339→   339→}

   341→   341→/** Append a single new event to the existing message list (immutable). */
   342→   342→function appendEvent(existing: ChatMessage[], ev: AgentEvent): ChatMessage[] {
   343→   343→  // The simplest correct merge: re-derive messages from existing events + the
   344→   344→  // new one. The existing messages were derived from a contiguous event
   345→   345→  // stream, so we just need their underlying events back... but we don't have
   346→   346→  // those. Instead, do an efficient targeted update.
   347→   347→  const ts = (ev.ts || Date.now() / 1000) * 1000;
   348→   348→  const seq = ev.i ?? existing.length;
   349→   349→  const out = existing.slice();
   350→   350→
   351→   351→  switch (ev.type) {
   352→   352→    case "user": {
   353→   353→      // Backend echoes user messages. If we already have a pending user
   354→   354→      // message with the same content (the optimistic one we added locally),
   355→   355→      // mark it ack'd instead of duplicating.
   356→   356→      const idx = out.findIndex(
   357→   357→        (m) => m.role === "user" && m.pending && m.content === (ev.text || "")
   358→   358→      );
   359→   359→      if (idx !== -1) {
   360→   360→        out[idx] = { ...out[idx], pending: false, seq, timestamp: ts };
   361→   361→        return out;
   362→   362→      }
   363→   363→      // Otherwise, only add if we don't already have this exact seq as a user.
   364→   364→      if (!out.some((m) => m.role === "user" && m.seq === seq)) {
   365→   365→        out.push({
   366→   366→          id: genMsgId(),
   367→   367→          role: "user",
   368→   368→          content: ev.text || "",
   369→   369→          timestamp: ts,
   370→   370→          seq,
   371→   371→        });
   372→   372→      }
   373→   373→      return out;
   374→   374→    }
   375→   375→    case "assistant": {
   376→   376→      out.push({
   377→   377→        id: genMsgId(),
   378→   378→        role: "assistant",
   379→   379→        content: ev.text || "",
   380→   380→        timestamp: ts,
   381→   381→        seq,
   382→   382→      });
   383→   383→      return out;
   384→   384→    }
   385→   385→    case "assistant_delta": {
   386→   386→      // Find the last assistant message that is currently streaming.
   387→   387→      for (let i = out.length - 1; i >= 0; i--) {
   388→   388→        if (out[i].role === "assistant" && out[i].isStreaming) {
   389→   389→          out[i] = { ...out[i], content: out[i].content + (ev.text || ""), seq };
   390→   390→          return out;
   391→   391→        }
   392→   392→      }
   393→   393→      // No streaming assistant — start a new one.
   394→   394→      out.push({
   395→   395→        id: genMsgId(),
   396→   396→        role: "assistant",
   397→   397→        content: ev.text || "",
   398→   398→        isStreaming: true,
   399→   399→        timestamp: ts,
   400→   400→        seq,
   401→   401→      });
   402→   402→      return out;
   403→   403→    }
   404→   404→    case "thinking": {
   405→   405→      out.push({
   406→   406→        id: genMsgId(),
   407→   407→        role: "thinking",
   408→   408→        content: ev.text || "",
   409→   409→        timestamp: ts,
   410→   410→        seq,
   411→   411→      });
   412→   412→      return out;
   413→   413→    }
   414→   414→    case "thinking_delta": {
   415→   415→      for (let i = out.length - 1; i >= 0; i--) {
   416→   416→        if (out[i].role === "thinking" && out[i].isStreaming) {
   417→   417→          out[i] = { ...out[i], content: out[i].content + (ev.text || ""), seq };
   418→   418→          return out;
   419→   419→        }
   420→   420→      }
   421→   421→      out.push({
   422→   422→        id: genMsgId(),
   423→   423→        role: "thinking",
   424→   424→        content: ev.text || "",
   425→   425→        isStreaming: true,
   426→   426→        timestamp: ts,
   427→   427→        seq,
   428→   428→      });
   429→   429→      return out;
   430→   430→    }
   431→   431→    case "tool_use": {
   432→   432→      out.push({
   433→   433→        id: genMsgId(),
   434→   434→        role: "tool",
   435→   435→        content: "",
   436→   436→        toolName: ev.name || "tool",
   437→   437→        toolSummary: ev.summary || "",
   438→   438→        timestamp: ts,
   439→   439→        seq,
   440→   440→      });
   441→   441→      return out;
   442→   442→    }
   443→   443→    case "tool_result": {
   444→   444→      out.push({
   445→   445→        id: genMsgId(),
   446→   446→        role: "tool",
   447→   447→        content: ev.text || "",
   448→   448→        isError: ev.is_error,
   449→   449→        toolName: "result",
   450→   450→        timestamp: ts,
   451→   451→        seq,
   452→   452→      });
   453→   453→      return out;
   454→   454→    }
   455→   455→    case "status": {
   456→   456→      const st = ev as {
   457→   457→        state: AgentStatus;
   458→   458→        detail?: string;
   459→   459→        cost_usd?: number | null;
   460→   460→      };
   461→   461→      if (st.state === "idle" && st.detail === "interrupted") {
   462→   462→        out.push({
   463→   463→          id: genMsgId(),
   464→   464→          role: "status",
   465→   465→          content: "— stopped —",
   466→   466→          timestamp: ts,
   467→   467→          seq,
   468→   468→        });
   469→   469→      } else if (st.state === "error") {
   470→   470→        out.push({
   471→   471→          id: genMsgId(),
   472→   472→          role: "status",
   473→   473→          content: st.detail || "Agent error",
   474→   474→          isError: true,
   475→   475→          costUsd: st.cost_usd,
   476→   476→          timestamp: ts,
   477→   477→          seq,
   478→   478→        });
   479→   479→      }
   480→   480→      return out;
   481→   481→    }
   482→   482→    case "panel": {
   483→   483→      out.push({
   484→   484→        id: genMsgId(),
   485→   485→        role: "panel",
   486→   486→        content: ev.task_name || "Panel",
   487→   487→        panelStatus: ev.status || "starting",
   488→   488→        timestamp: ts,
   489→   489→        seq,
   490→   490→      });
   491→   491→      return out;
   492→   492→    }
   493→   493→    default:
   494→   494→      return out;
   495→   495→  }
   496→   496→}

   498→   498→/** Mark the last streaming message as done (called when a turn ends). */
   499→   499→function finalizeStreaming(existing: ChatMessage[]): ChatMessage[] {
   500→   500→  const out = existing.slice();
   501→   501→  for (let i = out.length - 1; i >= 0; i--) {
   502→   502→    if (out[i].isStreaming) {
   503→   503→      out[i] = { ...out[i], isStreaming: false };
   504→   504→      break;
   505→   505→    }
   506→   506→  }
   507→   507→  return out;
   508→   508→}

   726→   726→            /* ignore */

   822→   822→// ---------------------------------------------------------------------------
   823→   823→// Internal: run a single agent turn (called by sendMessage and the queue
   824→   824→// drainer).
   825→   825→// ---------------------------------------------------------------------------
   826→   826→
   827→   827→async function _runTurn(
   828→   828→  client: AgentClient,
   829→   829→  text: string,
   830→   830→  model: string | undefined,
   831→   831→  set: (partial: Partial<ChatState> | ((s: ChatState) => Partial<ChatState>)) => void,
   832→   832→  get: () => ChatState,
   833→   833→): Promise<void> {
   834→   834→  // Ensure we have an active chat session.
   835→   835→  let sessionId = get().activeSessionId;
   836→   836→  if (!sessionId) {
   837→   837→    try {
   838→   838→      sessionId = await get().createSession(client, model);
   839→   839→    } catch (e) {
   840→   840→      set({
   841→   841→        error: e instanceof Error ? e.message : "Failed to create session",
   842→   842→      });
   843→   843→      return;
   844→   844→    }
   845→   845→  }
   846→   846→
   847→   847→  // Add user message optimistically (pending until backend echoes it).
   848→   848→  const userMsg: ChatMessage = {
   849→   849→    id: genMsgId(),
   850→   850→    role: "user",
   851→   851→    content: text,
   852→   852→    timestamp: Date.now(),
   853→   853→    seq: -1,
   854→   854→    pending: true,
   855→   855→  };
   856→   856→  set((s) => ({
   857→   857→    messages: [...s.messages, userMsg],
   858→   858→    isBusy: true,
   859→   859→    isStreaming: true,
   860→   860→    error: null,
   861→   861→    status: "running",
   862→   862→    inputText: "",
   863→   863→  }));
   864→   864→
   865→   865→  // Buffer events for batch persistence at turn end.
   866→   866→  let bufferedEvents: AgentEvent[] = [];
   867→   867→  let since = 0;
   868→   868→  let agentSid: string | null = null;
   869→   869→
   870→   870→  try {
   871→   871→    const start: AgentStart = await client.send(
   872→   872→      text,
   873→   873→      undefined, // no session_id — the backend reuses by chat_session_id
   874→   874→      model || undefined,
   875→   875→      get().workspaceId || undefined,
   876→   876→      sessionId,
   877→   877→    );
   878→   878→    agentSid = start.session_id;
   879→   879→    since = 0;
   880→   880→    set({ _agentSessionId: agentSid });
   881→   881→
   882→   882→    // Open SSE stream for live events. If SSE fails, fall back to polling.
   883→   883→    await new Promise<void>((resolve) => {
   884→   884→      let settled = false;
   885→   885→      let pollTimer: ReturnType<typeof setInterval> | null = null;
   886→   886→      const finish = () => {
   887→   887→        if (settled) return;
   888→   888→        settled = true;
   889→   889→        if (pollTimer) clearInterval(pollTimer);
   890→   890→        resolve();
   891→   891→      };
   892→   892→
   893→   893→      const handleEvent = (ev: AgentEvent) => {
   894→   894→        bufferedEvents.push(ev);
   895→   895→        since = Math.max(since, (ev.i ?? 0) + 1);
   896→   896→        set((s) => ({ messages: appendEvent(s.messages, ev) }));
   897→   897→
   898→   898→        // Update status from status events.
   899→   899→        if (ev.type === "status") {
   900→   900→          const st = ev as { state: AgentStatus; cost_usd?: number | null };
   901→   901→          set({ status: st.state });
   902→   902→          if (typeof st.cost_usd === "number") {
   903→   903→            set({ cost: st.cost_usd });
   904→   904→          }
   905→   905→          if (st.state === "idle" || st.state === "error") {
   906→   906→            set((s) => ({ messages: finalizeStreaming(s.messages) }));
   907→   907→            // Give a brief moment for any trailing events, then finish.
   908→   908→            setTimeout(finish, 200);
   909→   909→          }
   910→   910→        }
   911→   911→      };
   912→   912→
   913→   913→      const controller = client.stream(
   914→   914→        agentSid!,
   915→   915→        since,
   916→   916→        handleEvent,
   917→   917→        () => {
   918→   918→          // SSE failed — fall back to polling.
   919→   919→          if (pollTimer) return; // already polling
   920→   920→          pollTimer = setInterval(async () => {
   921→   921→            try {
   922→   922→              const snap = await client.poll(agentSid!, since);
   923→   923→              for (const ev of snap.events) handleEvent(ev);
   924→   924→              if (snap.status !== "running" && snap.status !== "starting") {
   925→   925→                set((s) => ({ messages: finalizeStreaming(s.messages) }));
   926→   926→                finish();
   927→   927→              }
   928→   928→            } catch {
   929→   929→              /* keep trying */
   930→   930→            }
   931→   931→          }, 800);
   932→   932→        },
   933→   933→        () => {
   934→   934→          // Stream closed by server — turn done (or socket dropped).
   935→   935→          // Do one final poll to catch any missed events.
   936→   936→          client
   937→   937→            .poll(agentSid!, since)
   938→   938→            .then((snap) => {
   939→   939→              for (const ev of snap.events) handleEvent(ev);
   940→   940→              set((s) => ({ messages: finalizeStreaming(s.messages) }));
   941→   941→              finish();
   942→   942→            })
   943→   943→            .catch(() => finish());
   944→   944→        },
   945→   945→      );
   946→   946→      set({ _streamController: controller });
   947→   947→    });
   948→   948→
   949→   949→    // Persist all buffered events to the chat session (backend dedupes).
   950→   950→    if (bufferedEvents.length > 0) {
   951→   951→      await client.persistEvents(sessionId, bufferedEvents);
   952→   952→    }
   953→   953→
   954→   954→    // Refresh files (best effort).
   955→   955→    try {
   956→   956→      const f = await client.files(agentSid);
   957→   957→      set({ files: f.files });
   958→   958→    } catch {
   959→   959→      /* ignore */
   960→   960→    }
   961→   961→
   962→   962→    // Refresh session list to pick up auto-title.
   963→   963→    try {
   964→   964→      const { sessions } = await client.listChatSessions();
   965→   965→      set({ sessions });
   966→   966→    } catch {
   967→   967→      /* ignore */
   968→   968→    }
   969→   969→
   970→   970→    set({
   971→   971→      isBusy: false,
   972→   972→      isStreaming: false,
   973→   973→      _agentSessionId: null,
   974→   974→      _streamController: null,
   975→   975→    });
   976→   976→  } catch (e) {
   977→   977→    set({
   978→   978→      error: e instanceof Error ? e.message : String(e),
   979→   979→      isBusy: false,
   980→   980→      isStreaming: false,
   981→   981→      status: "error",
   982→   982→      messages: finalizeStreaming(get().messages),
   983→   983→      _agentSessionId: null,
   984→   984→      _streamController: null,
   985→   985→    });
   986→   986→    return;
   987→   987→  }
   988→   988→
   989→   989→  // Drain the queue: send the next message if any.
   990→   990→  const next = get().queue[0];
   991→   991→  if (next) {
   992→   992→    set((s) => ({ queue: s.queue.slice(1) }));
   993→   993→    await _runTurn(client, next.text, model, set, get);
   994→   994→  }
   995→   995→}
   996→   996→
   997→   997→// On store creation, restore workspaceId from localStorage (best effort).
   998→   998→try {
   999→   999→  const stored = localStorage.getItem(WORKSPACE_KEY);
  1000→  1000→  if (stored) {
  1001→  1001→    useChatStore.setState({ workspaceId: stored });
  1002→  1002→  }
  1003→  1003→} catch {
  1004→  1004→  /* ignore */
  1005→  1005→}
  1006→  1006→
