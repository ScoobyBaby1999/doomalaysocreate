     1→import { useEffect, useRef, useCallback, useState, useMemo } from "react";
     2→import type { Settings } from "../api/panel";
     3→import { AgentClient } from "../api/agent";
     4→import { useChatStore } from "../state/chatStore";
     5→import { useModelStore } from "../lib/model-store";
     6→import { ChatMessageBubble } from "../components/ChatMessageBubble";
     7→import { SessionSidebar } from "../components/SessionSidebar";
     8→import { FileDrawer } from "../components/FileDrawer";
     9→import { PanelDrawer, type PanelInvocation } from "../components/PanelDrawer";
    10→
    11→export function AgentChat({ settings }: { settings: Settings }) {
    12→  // Model store
    13→  const openOverlay = useModelStore((s) => s.openOverlay);
    14→  const selectedModelId = useModelStore((s) => s.selectedModelId);
    15→  const selectedProviderName = useModelStore((s) => s.selectedProviderName);
    16→  const providers = useModelStore((s) => s.providers);
    17→  const selectedSlotId = useModelStore((s) => s.selectedSlotId);
    18→
    19→  // Derive display info for selected model
    20→  const modelDisplay = (() => {
    21→    if (!selectedModelId) return null;
    22→    for (const p of providers) {
    23→      const m = p.models.find(
    24→        (m) => m.id === selectedModelId || m.slotId === selectedSlotId,
    25→      );
    26→      if (m)
    27→        return {
    28→          label: m.displayName || m.id,
    29→          color: p.color || "#5b8cff",
    30→        };
    31→    }
    32→    return {
    33→      label: selectedModelId.split("/").pop() || selectedModelId,
    34→      color: "#5b8cff",
    35→    };
    36→  })();
    37→
    38→  // The model ID to send to the backend. Prefer the physical slot ID
    39→  // ("provider/model"), fall back to the logical ID. The backend resolves
    40→  // either form (see _handle_agent_post).
    41→  const effectiveModelId = selectedSlotId || selectedModelId;
    42→
    43→  // Chat store — use SELECTIVE subscriptions so typing in the input doesn't
    44→  // re-render the whole message list. Each useChatStore((s) => ...) only
    45→  // re-renders when that specific slice changes.
    46→  const sessions = useChatStore((s) => s.sessions);
    47→  const activeSessionId = useChatStore((s) => s.activeSessionId);
    48→  const messages = useChatStore((s) => s.messages);
    49→  const isBusy = useChatStore((s) => s.isBusy);
    50→  const isStreaming = useChatStore((s) => s.isStreaming);
    51→  const status = useChatStore((s) => s.status);
    52→  const inputText = useChatStore((s) => s.inputText);
    53→  const error = useChatStore((s) => s.error);
    54→  const effort = useChatStore((s) => s.effort);
    55→  const webSearch = useChatStore((s) => s.webSearch);
    56→  const deepResearch = useChatStore((s) => s.deepResearch);
    57→  const files = useChatStore((s) => s.files);
    58→  const fileDrawerOpen = useChatStore((s) => s.fileDrawerOpen);
    59→  const panelDrawerOpen = useChatStore((s) => s.panelDrawerOpen);
    60→  const panelInvocations = useChatStore((s) => s.panelInvocations);
    61→  const sidebarOpen = useChatStore((s) => s.sidebarOpen);
    62→  const cost = useChatStore((s) => s.cost);
    63→  const queue = useChatStore((s) => s.queue);
    64→  const isLoadingMessages = useChatStore((s) => s.isLoadingMessages);
    65→  // Model verification fields
    66→  const resolvedModel = useChatStore((s) => s.resolvedModel);
    67→  const resolvedProvider = useChatStore((s) => s.resolvedProvider);
    68→  const requestedModel = useChatStore((s) => s.requestedModel);
    69→  // Actions (stable references from zustand — don't cause re-renders)
    70→  const setInputText = useChatStore((s) => s.setInputText);
    71→  const setEffort = useChatStore((s) => s.setEffort);
    72→  const toggleWebSearch = useChatStore((s) => s.toggleWebSearch);
    73→  const toggleDeepResearch = useChatStore((s) => s.toggleDeepResearch);
    74→  const setSidebarOpen = useChatStore((s) => s.setSidebarOpen);
    75→  const setFileDrawerOpen = useChatStore((s) => s.setFileDrawerOpen);
    76→  const setPanelDrawerOpen = useChatStore((s) => s.setPanelDrawerOpen);
    77→  const loadSessions = useChatStore((s) => s.loadSessions);
    78→  const createSession = useChatStore((s) => s.createSession);
    79→  const switchSession = useChatStore((s) => s.switchSession);
    80→  const deleteSession = useChatStore((s) => s.deleteSession);
    81→  const renameSession = useChatStore((s) => s.renameSession);
    82→  const sendMessage = useChatStore((s) => s.sendMessage);
    83→  const stopGeneration = useChatStore((s) => s.stopGeneration);
    84→  const dequeueMessage = useChatStore((s) => s.dequeueMessage);
    85→
    86→  // Refs
    87→  const clientRef = useRef(new AgentClient(settings));
    88→  const scrollRef = useRef<HTMLDivElement>(null);
    89→  const inputRef = useRef<HTMLTextAreaElement>(null);
    90→  const lastUserMsgRef = useRef("");
    91→  const stickToBottomRef = useRef(true);
    92→
    93→  // Keep client fresh when settings change — useMemo so it only rebuilds
    94→  // when settings actually change, NOT on every render.
    95→  const client = useMemo(() => new AgentClient(settings), [settings]);
    96→  clientRef.current = client;
    97→
    98→  // Auto-scroll behavior: only stick to bottom if the user is already there.
    99→  const handleScroll = useCallback(() => {
   100→    const el = scrollRef.current;
   101→    if (!el) return;
   102→    const atBottom =
   103→      el.scrollHeight - el.scrollTop - el.clientHeight < 80;
   104→    stickToBottomRef.current = atBottom;
   105→  }, []);
   106→
   107→  useEffect(() => {
   108→    if (scrollRef.current && stickToBottomRef.current) {
   109→      scrollRef.current.scrollTop = scrollRef.current.scrollHeight;
   110→    }
   111→  }, [messages, isStreaming]);
   112→
   113→  // Load sessions on mount (auto-restores active session if any).
   114→  useEffect(() => {
   115→    loadSessions(clientRef.current);
   116→    // eslint-disable-next-line react-hooks/exhaustive-deps
   117→  }, []);
   118→
   119→  // Handle send — note the store handles queueing when busy.
   120→  const handleSend = useCallback(
   121→    async (text?: string) => {
   122→      const msg = (text ?? inputText).trim();
   123→      if (!msg) return;
   124→      lastUserMsgRef.current = msg;
   125→      await sendMessage(clientRef.current, msg, effectiveModelId || undefined);
   126→    },
   127→    [inputText, effectiveModelId, sendMessage],
   128→  );
   129→
   130→  // Handle stop
   131→  const handleStop = useCallback(async () => {
   132→    await stopGeneration(clientRef.current);
   133→  }, [stopGeneration]);
   134→
   135→  // Handle new session
   136→  const handleNewSession = useCallback(async () => {
   137→    await createSession(clientRef.current, effectiveModelId || undefined);
   138→    inputRef.current?.focus();
   139→  }, [createSession, effectiveModelId]);
   140→
   141→  // Handle keydown — Enter sends, Shift+Enter newlines. While busy, Enter
   142→  // queues instead of being disabled.
   143→  const handleKeyDown = useCallback(
   144→    (e: React.KeyboardEvent<HTMLTextAreaElement>) => {
   145→      if (e.key === "Enter" && !e.shiftKey) {
   146→        e.preventDefault();
   147→        handleSend();
   148→      }
   149→    },
   150→    [handleSend],
   151→  );
   152→
   153→  // Auto-resize textarea
   154→  const handleInputChange = useCallback(
   155→    (e: React.ChangeEvent<HTMLTextAreaElement>) => {
   156→      setInputText(e.target.value);
   157→      const el = e.target;
   158→      el.style.height = "auto";
   159→      el.style.height = Math.min(el.scrollHeight, 160) + "px";
   160→    },
   161→    [setInputText],
   162→  );
   163→
   164→  // Settings panel
   165→  const [settingsOpen, setSettingsOpen] = useState(false);
   166→
   167→  const running = isBusy && (status === "running" || status === "starting");
   168→  const queueCount = queue.length;
   169→
   170→  return (
   171→    <div className="flex flex-col h-full relative bg-bg">
   172→      {/* Header */}
   173→      <header className="flex items-center gap-2 px-3 h-11 border-b border-border shrink-0 bg-surface/50">
   174→        {/* Session menu toggle */}
   175→        <button
   176→          onClick={() => setSidebarOpen(!sidebarOpen)}
   177→          className={`p-1.5 rounded-md transition-colors ${
   178→            sidebarOpen
   179→              ? "text-accent bg-accent/10"
   180→              : "text-muted-foreground hover:text-foreground hover:bg-surface2"
   181→          }`}
   182→          title="Chat sessions"
   183→        >
   184→          <svg width="16" height="16" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round">
   185→            <line x1="3" y1="6" x2="21" y2="6" />
   186→            <line x1="3" y1="12" x2="21" y2="12" />
   187→            <line x1="3" y1="18" x2="21" y2="18" />
   188→          </svg>
   189→        </button>
   190→
   191→        {/* Active session title (truncated) */}
   192→        <div className="text-[12.5px] font-medium text-foreground truncate max-w-[140px]">
   193→          {activeSessionId
   194→            ? sessions.find((s) => s.id === activeSessionId)?.title || "New Chat"
   195→            : "New Chat"}
   196→        </div>
   197→
   198→        {/* Model selector */}
   199→        <button
   200→          onClick={() => !running && openOverlay()}
   201→          disabled={running}
   202→          className="flex items-center gap-1.5 px-2.5 py-1 rounded-lg border border-border hover:border-accent transition-colors text-[11.5px] disabled:opacity-50 disabled:cursor-not-allowed max-w-[200px] ml-auto"
   203→          title={selectedProviderName || "Select model"}
   204→        >
   205→          <span
   206→            className="w-2 h-2 rounded-full shrink-0"
   207→            style={{ background: modelDisplay?.color || "#5b8cff" }}
   208→          />
   209→          <span className="truncate text-accent">
   210→            {modelDisplay?.label || "Select model"}
   211→          </span>
   212→          <svg width="10" height="10" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2.5" strokeLinecap="round" strokeLinejoin="round" className="text-muted-foreground shrink-0">
   213→            <polyline points="6 9 12 15 18 9" />
   214→          </svg>
   215→        </button>
   216→
   217→        {/* Model verification badge — shows what the backend ACTUALLY resolved.
   218→            If resolvedModel differs from requestedModel, show a warning so the
   219→            user knows the model they picked isn't the one running. */}
   220→        {resolvedModel && (
   221→          <div className="flex flex-col items-end gap-0.5 ml-1" title={`Verified: ${resolvedProvider || "?"} → ${resolvedModel}`}>
   222→            <span className="text-[9px] text-emerald-400/80 font-mono tabular-nums flex items-center gap-0.5">
   223→              <svg width="8" height="8" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="3" strokeLinecap="round" strokeLinejoin="round"><polyline points="20 6 9 17 4 12"/></svg>
   224→              {resolvedProvider || "unknown"}
   225→            </span>
   226→            {requestedModel && resolvedModel &&
   227→             requestedModel.split("/").pop()?.toLowerCase() !== resolvedModel.split("/").pop()?.toLowerCase() && (
   228→              <span className="text-[8px] text-amber-400/90 font-mono" title={`Requested ${requestedModel} but backend resolved to ${resolvedModel}`}>
   229→                ⚠ redirected
   230→              </span>
   231→            )}
   232→          </div>
   233→        )}
   234→
   235→        {/* Cost indicator */}
   236→        {cost != null && cost > 0 && (
   237→          <span className="text-[10px] text-muted-foreground tabular-nums">
   238→            ${cost.toFixed(4)}
   239→          </span>
   240→        )}
   241→
   242→        {/* Status pill */}
   243→        {running && (
   244→          <div className="flex items-center gap-1.5 text-[10.5px] text-muted-foreground">
   245→            <span className="w-1.5 h-1.5 rounded-full bg-accent animate-pulse" />
   246→            <span className="capitalize">{status}</span>
   247→          </div>
   248→        )}
   249→
   250→        {/* Files */}
   251→        <button
   252→          onClick={() => setFileDrawerOpen(!fileDrawerOpen)}
   253→          className={`p-1.5 rounded-md transition-colors ${
   254→            fileDrawerOpen
   255→              ? "text-accent bg-accent/10"
   256→              : "text-muted-foreground hover:text-foreground hover:bg-surface2"
   257→          }`}
   258→          title="Files"
   259→        >
   260→          <svg width="16" height="16" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round">
   261→            <path d="M13.5 2H6a2 2 0 0 0-2 2v16a2 2 0 0 0 2 2h12a2 2 0 0 0 2-2V9z" />
   262→            <polyline points="13 2 13 9 20 9" />
   263→          </svg>
   264→        </button>
   265→
   266→        <button
   267→          onClick={() => setPanelDrawerOpen(!panelDrawerOpen)}
   268→          className={`p-1.5 rounded-md transition-colors ${
   269→            panelDrawerOpen
   270→              ? "text-accent bg-accent/10"
   271→              : "text-muted-foreground hover:text-foreground hover:bg-surface2"
   272→          }`}
   273→          title="Panel invocations"
   274→        >
   275→          <svg width="16" height="16" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round">
   276→            <path d="M21 15a2 2 0 0 1-2 2H7l-4 4V5a2 2 0 0 1 2-2h14a2 2 0 0 1 2 2z" />
   277→          </svg>
   278→        </button>
   279→
   280→        {running ? (
   281→          <button
   282→            onClick={handleStop}
   283→            className="flex items-center gap-1 text-[11px] px-2.5 py-1.5 rounded-md bg-red-500/15 text-red-300 hover:bg-red-500/25 transition-colors font-medium shrink-0"
   284→          >
   285→            <svg width="11" height="11" viewBox="0 0 24 24" fill="currentColor">
   286→              <rect x="6" y="6" width="12" height="12" rx="1" />
   287→            </svg>
   288→            Stop
   289→          </button>
   290→        ) : (
   291→          <button
   292→            onClick={handleNewSession}
   293→            className="flex items-center gap-1 text-[11px] px-2.5 py-1.5 rounded-md bg-accent/15 text-accent hover:bg-accent/25 transition-colors font-medium shrink-0"
   294→          >
   295→            <svg width="11" height="11" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2.5" strokeLinecap="round" strokeLinejoin="round">
   296→              <line x1="12" y1="5" x2="12" y2="19" />
   297→              <line x1="5" y1="12" x2="19" y2="12" />
   298→            </svg>
   299→            New
   300→          </button>
   301→        )}
   302→      </header>
   303→
   304→      {/* Messages area */}
   305→      <div
   306→        ref={scrollRef}
   307→        onScroll={handleScroll}
   308→        className="flex-1 overflow-y-auto min-h-0 scroll-smooth"
   309→      >
   310→        {isLoadingMessages && messages.length === 0 ? (
   311→          <div className="flex items-center justify-center h-full">
   312→            <div className="flex items-center gap-2 text-muted-foreground text-[12px]">
   313→              <div className="w-3 h-3 border-2 border-accent border-t-transparent rounded-full animate-spin" />
   314→              Loading chat…
   315→            </div>
   316→          </div>
   317→        ) : messages.length === 0 && !isBusy ? (
   318→          <EmptyState onSend={handleSend} />
   319→        ) : (
   320→          <div className="py-3 max-w-3xl mx-auto">
   321→            {messages.map((msg) => (
   322→              <ChatMessageBubble key={msg.id} message={msg} />
   323→            ))}
   324→
   325→            {/* Typing indicator — only when the latest message is a user
   326→                message and we haven't received the first delta yet. */}
   327→            {running &&
   328→              messages.length > 0 &&
   329→              messages[messages.length - 1].role === "user" && (
   330→                <div className="px-4 py-2.5">
   331→                  <div className="flex items-center gap-2.5">
   332→                    <div className="shrink-0 w-7 h-7 rounded-md bg-accent/15 flex items-center justify-center">
   333→                      <svg width="14" height="14" viewBox="0 0 24 24" fill="none" stroke="#5b8cff" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round">
   334→                        <path d="M12 2a10 10 0 0 1 10 10c0 5.523-4.477 10-10 10S2 17.523 2 12 6.477 2 12 2z" />
   335→                        <path d="M12 16v-4" />
   336→                        <path d="M12 8h.01" />
   337→                      </svg>
   338→                    </div>
   339→                    <span className="flex gap-1 items-center h-4">
   340→                      <span className="w-1.5 h-1.5 rounded-full bg-muted-foreground animate-bounce" style={{ animationDelay: "0ms" }} />
   341→                      <span className="w-1.5 h-1.5 rounded-full bg-muted-foreground animate-bounce" style={{ animationDelay: "150ms" }} />
   342→                      <span className="w-1.5 h-1.5 rounded-full bg-muted-foreground animate-bounce" style={{ animationDelay: "300ms" }} />
   343→                    </span>
   344→                  </div>
   345→                </div>
   346→              )}
   347→          </div>
   348→        )}
   349→      </div>
   350→
   351→      {/* Queue indicator (1-up: shows pending messages while LLM is busy) */}
   352→      {queueCount > 0 && (
   353→        <div className="border-t border-border bg-accent/5 px-3 py-2 shrink-0">
   354→          <div className="flex items-center gap-2 text-[11.5px] text-accent">
   355→            <svg width="12" height="12" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" className="animate-pulse">
   356→              <circle cx="12" cy="12" r="10" /><polyline points="12 6 12 12 16 14" />
   357→            </svg>
   358→            <span className="font-medium">
   359→              {queueCount} message{queueCount !== 1 ? "s" : ""} queued
   360→            </span>
   361→            <div className="flex-1 truncate text-muted-foreground/80">
   362→              Next: {queue[0]?.text.slice(0, 80)}
   363→              {queue[0] && queue[0].text.length > 80 ? "…" : ""}
   364→            </div>
   365→            <button
   366→              onClick={() => queue.forEach((q) => dequeueMessage(q.id))}
   367→              className="text-muted-foreground hover:text-foreground px-1.5 py-0.5 rounded hover:bg-surface2 transition-colors"
   368→              title="Clear queue"
   369→            >
   370→              Clear
   371→            </button>
   372→          </div>
   373→        </div>
   374→      )}
   375→
   376→      {/* Error banner */}
   377→      {error && (
   378→        <div className="border-t border-red-500/30 bg-red-500/10 px-4 py-2.5 text-[13px] text-red-300 flex items-center gap-3 shrink-0">
   379→          <svg width="14" height="14" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" className="shrink-0">
   380→            <circle cx="12" cy="12" r="10" /><line x1="12" y1="8" x2="12" y2="12" /><line x1="12" y1="16" x2="12.01" y2="16" />
   381→          </svg>
   382→          <span className="flex-1">{error}</span>
   383→          <button
   384→            onClick={() => useChatStore.setState({ error: null })}
   385→            className="text-red-300/70 hover:text-red-200 px-2 py-0.5 text-[11px]"
   386→          >
   387→            dismiss
   388→          </button>
   389→          {lastUserMsgRef.current && (
   390→            <button
   391→              onClick={() => handleSend(lastUserMsgRef.current)}
   392→              className="px-3 py-1 rounded-md border border-red-500/40 text-[11px] hover:bg-red-500/20 transition-colors shrink-0"
   393→            >
   394→              Retry
   395→            </button>
   396→          )}
   397→        </div>
   398→      )}
   399→
   400→      {/* Input area */}
   401→      <div className="border-t border-border bg-surface/30 pt-2 pb-3 px-3 shrink-0">
   402→        {/* Quick settings */}
   403→        {settingsOpen && (
   404→          <div className="flex items-center gap-1.5 px-1 pb-2 overflow-x-auto">
   405→            <span className="text-[10px] uppercase tracking-wide text-muted-foreground/60 mr-1 shrink-0">Effort</span>
   406→            {(["low", "med", "high", "max"] as const).map((e) => (
   407→              <button
   408→                key={e}
   409→                onClick={() => setEffort(e)}
   410→                className={`text-[11px] px-2.5 py-1 rounded-full border transition-colors shrink-0 ${
   411→                  effort === e
   412→                    ? "border-accent text-accent bg-accent/10"
   413→                    : "border-border text-muted-foreground hover:border-accent/50"
   414→                }`}
   415→              >
   416→                {e}
   417→              </button>
   418→            ))}
   419→            <div className="w-px h-4 bg-border mx-1 shrink-0" />
   420→            <button
   421→              onClick={toggleWebSearch}
   422→              className={`text-[11px] px-2.5 py-1 rounded-full border transition-colors shrink-0 flex items-center gap-1 ${
   423→                webSearch
   424→                  ? "border-accent text-accent bg-accent/10"
   425→                  : "border-border text-muted-foreground hover:border-accent/50"
   426→              }`}
   427→            >
   428→              <svg width="11" height="11" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round">
   429→                <circle cx="12" cy="12" r="10" /><line x1="2" y1="12" x2="22" y2="12" /><path d="M12 2a15.3 15.3 0 0 1 4 10 15.3 15.3 0 0 1-4 10 15.3 15.3 0 0 1-4-10 15.3 15.3 0 0 1 4-10z" />
   430→              </svg>
   431→              Web
   432→            </button>
   433→            <button
   434→              onClick={toggleDeepResearch}
   435→              className={`text-[11px] px-2.5 py-1 rounded-full border transition-colors shrink-0 flex items-center gap-1 ${
   436→                deepResearch
   437→                  ? "border-accent text-accent bg-accent/10"
   438→                  : "border-border text-muted-foreground hover:border-accent/50"
   439→              }`}
   440→            >
   441→              <svg width="11" height="11" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round">
   442→                <path d="M12 22s-8-4.5-8-11.8A8 8 0 0 1 12 2a8 8 0 0 1 8 8.2c0 7.3-8 11.8-8 11.8z" /><circle cx="12" cy="10" r="3" />
   443→              </svg>
   444→              Deep
   445→            </button>
   446→          </div>
   447→        )}
   448→
   449→        {/* Input row */}
   450→        <div className="flex items-end gap-2 max-w-3xl mx-auto">
   451→          <button
   452→            onClick={() => setSettingsOpen((p) => !p)}
   453→            className={`p-2 rounded-lg transition-colors shrink-0 mb-0.5 ${
   454→              settingsOpen
   455→                ? "text-accent bg-accent/10"
   456→                : "text-muted-foreground hover:text-foreground hover:bg-surface2"
   457→            }`}
   458→            title="Settings"
   459→          >
