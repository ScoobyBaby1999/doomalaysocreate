    54→function syncedAgoLabel(iso: string | null): string {
    55→  if (!iso) return "";
    56→  const then = new Date(iso).getTime();
    57→  if (Number.isNaN(then)) return "";
    58→  const secs = Math.max(0, Math.round((Date.now() - then) / 1000));
    59→  if (secs < 5) return "synced just now";
    60→  if (secs < 60) return `synced ${secs}s ago`;
    61→  const mins = Math.round(secs / 60);
    62→  if (mins < 60) return `synced ${mins}m ago`;
    63→  const hrs = Math.round(mins / 60);
    64→  return `synced ${hrs}h ago`;
    65→}

    87→function modelMatchesFilters(model: ProviderModel, activeFilters: string[], contextMin: number): boolean {
    88→  const caps = model.attributes?.capabilities ?? [];
    89→  const bm = model.attributes?.benchmarks;
    90→
    91→  if (contextMin > 0 && model.contextLength < contextMin) return false;
    92→  if (activeFilters.length === 0) return true;
    93→
    94→  return activeFilters.some((filter) => {
    95→    if (filter === "reasoning") {
    96→      if (caps.includes("reasoning")) return true;
    97→      if ((bm?.intelligence ?? 0) >= 20) return true;
    98→    }
    99→    if (filter === "code") {
   100→      if (caps.includes("code")) return true;
   101→      if ((bm?.coding ?? 0) >= 20 || (bm?.aaCoding ?? 0) >= 20) return true;
   102→    }
   103→    if (filter === "tools") {
   104→      if (caps.includes("tools") || caps.includes("tool use") || caps.includes("function calling")) return true;
   105→      if ((bm?.agentic ?? 0) >= 20) return true;
   106→    }
   107→    if (filter === "vision") {
   108→      if (caps.includes("vision")) return true;
   109→    }
   110→    if (filter === "speech") {
   111→      if (caps.includes("speech") || caps.includes("audio")) return true;
   112→    }
   113→    return false;
   114→  });
   115→}

   117→function isReasoningModel(model: { displayName: string; id?: string; logical?: string; attributes?: ModelAttributes }): boolean {
   118→  const caps = model.attributes?.capabilities ?? [];
   119→  if (caps.includes("reasoning")) return true;

   124→function bestFilterScore(model: ProviderModel, activeFilters: string[]): number {
   125→  const bm = model.attributes?.benchmarks;
   126→  let best = 0;
   127→  for (const filter of activeFilters) {
   128→    if (filter === "reasoning") best = Math.max(best, isReasoningModel(model) ? 1 : 0);
   129→    if (filter === "intelligence") best = Math.max(best, bm?.intelligence ?? 0);
   130→    if (filter === "code") best = Math.max(best, bm?.coding ?? 0, bm?.aaCoding ?? 0);
   131→    if (filter === "agent") best = Math.max(best, bm?.agentic ?? 0);
   132→    if (filter === "tools") best = Math.max(best, bm?.agentic ?? 0);
   133→  }
   134→  return best;
   135→}

   137→function sortByFilterScore(models: ProviderModel[], activeFilters: string[]): ProviderModel[] {
   138→  if (activeFilters.length === 0) return models;
   139→  return [...models].sort((a, b) => bestFilterScore(b, activeFilters) - bestFilterScore(a, activeFilters));
   140→}

   142→function avgRank(attrs: ModelAttributes | undefined): number {
   143→  const ranks = attrs?.ranks;
   144→  if (!ranks || ranks.length === 0) return Infinity;
   145→  return ranks.reduce((s, r) => s + r.rank, 0) / ranks.length;
   146→}
   147→
   148→function defaultSort(models: ProviderModel[]): ProviderModel[] {
   149→  return [...models].sort((a, b) => {
   150→    const aAvg = avgRank(a.attributes);
   151→    const bAvg = avgRank(b.attributes);
   152→    if (aAvg !== bAvg) return aAvg === Infinity ? 1 : bAvg === Infinity ? -1 : aAvg - bAvg;
   153→
   154→    const aIntel = a.attributes?.benchmarks?.intelligence ?? 0;
   155→    const bIntel = b.attributes?.benchmarks?.intelligence ?? 0;
   156→    if (aIntel !== bIntel) return bIntel - aIntel;
   157→
   158→    const aCode = a.attributes?.benchmarks?.coding ?? 0;
   159→    const bCode = b.attributes?.benchmarks?.coding ?? 0;
   160→    if (aCode !== bCode) return bCode - aCode;
   161→
   162→    return 0;
   163→  });
   164→}

   188→function attributeSegments(a: ModelAttributes | undefined): AttrSegment[] | null {
   189→  if (!a) return null;
   190→  const segs: AttrSegment[] = [];
   191→  const b = a.benchmarks;
   192→  if (b) {
   193→    if (typeof b.intelligence === "number") {
   194→      if (segs.length) segs.push({ text: " \u00b7 " });
   195→      segs.push({ text: `AA ${b.intelligence}`, color: scoreColor(b.intelligence) });
   196→    }
   197→    if (typeof b.coding === "number") {
   198→      if (segs.length) segs.push({ text: " \u00b7 " });
   199→      segs.push({ text: `code ${b.coding}`, color: scoreColor(b.coding) });
   200→    }
   201→    if (typeof b.agentic === "number") {
   202→      if (segs.length) segs.push({ text: " \u00b7 " });
   203→      segs.push({ text: `agent ${b.agentic}`, color: scoreColor(b.agentic) });
   204→    }
   205→    if (typeof b.sweBench === "number") {
   206→      if (segs.length) segs.push({ text: " \u00b7 " });
   207→      segs.push({ text: `SWE ${b.sweBench}`, color: scoreColor(b.sweBench) });
   208→    }
   209→    if (typeof b.aaCoding === "number" && typeof b.sweBench === "undefined") {
   210→      if (segs.length) segs.push({ text: " \u00b7 " });
   211→      segs.push({ text: `AA-coding ${b.aaCoding}`, color: scoreColor(b.aaCoding) });
   212→    }
   213→  }
   214→  if (a.ranks && a.ranks.length > 0) {
   215→    const top = a.ranks.slice(0, 3).map((r) => `${r.label} #${r.rank}`).join(" \u00b7 ");
   216→    if (segs.length) segs.push({ text: " \u00b7 " });
   217→    segs.push({ text: top });
   218→  }
   219→  if (a.capabilities && a.capabilities.length > 0) {
   220→    for (const cap of a.capabilities) {
   221→      if (segs.length) segs.push({ text: " \u00b7 " });
   222→      segs.push({ text: cap, color: capColor(cap) });
   223→    }
   224→  }
   225→  if (a.pricing) {
   226→    if (segs.length) segs.push({ text: " \u00b7 " });
   227→    segs.push({ text: a.pricing });
   228→  }
   229→
   230→  if (segs.length === 0 && !a.note) return null;
   231→  return segs;
   232→}

   311→function ProviderBox({
   312→  provider,
   313→  selectedModelId,
   314→  selectedProviderName,
   315→  condensedView,
   316→  onSelect,
   317→  searchQuery,
   318→  activeFilters,
   319→  contextMin,
   320→}: {
   321→  provider: ProviderGroup;
   322→  selectedModelId: string | null;
   323→  selectedProviderName: string | null;
   324→  condensedView: boolean;
   325→  onSelect: (model: ProviderModel, providerName: string) => void;
   326→  searchQuery: string;
   327→  activeFilters: string[];
   328→  contextMin: number;
   329→}) {
   330→  const [showDimmed, setShowDimmed] = useState(false);
   331→
   332→  const { matched, dimmed } = useMemo(() => {
   333→    const allModels = provider.models;
   334→    const searched = !searchQuery.trim()
   335→      ? allModels
   336→      : allModels.filter((m) => {
   337→          const q = searchQuery.toLowerCase();
   338→          return (
   339→            m.displayName.toLowerCase().includes(q) ||
   340→            m.id.toLowerCase().includes(q) ||
   341→            (m.family?.toLowerCase().includes(q) ?? false)
   342→          );
   343→        });
   344→    const m: ProviderModel[] = [];
   345→    const d: ProviderModel[] = [];
   346→    for (const mdl of searched) {
   347→      if (modelMatchesFilters(mdl, activeFilters, contextMin)) {
   348→        m.push(mdl);
   349→      } else {
   350→        d.push(mdl);
   351→      }
   352→    }
   353→    return {
   354→      matched: activeFilters.length > 0 || contextMin > 0 ? sortByFilterScore(m, activeFilters) : defaultSort(m),
   355→      dimmed: d,
   356→    };
   357→  }, [provider.models, searchQuery, activeFilters, contextMin]);
   358→
   359→  const openSettings = useCallback(() => {
   360→    useModelStore.getState().openProvidersDialog(provider.name);
   361→  }, [provider.name]);
   362→
   363→  const hasDimmed = dimmed.length > 0;
   364→  const empty = matched.length === 0 && !hasDimmed;
   365→
   366→  return (
   367→    <div
   368→      className="flex flex-col rounded-lg border overflow-hidden"
   369→      style={{ borderColor: `${provider.color}30` }}
   370→    >
   371→      <div
   372→        className="flex items-center gap-2 px-2.5 shrink-0"
   373→        style={{ height: 26, backgroundColor: `${provider.color}0d`, borderBottom: `1px solid ${provider.color}1f` }}

   375→        <span className="inline-block w-2 h-2 rounded-full shrink-0" style={{ backgroundColor: provider.color }} />
   376→        <span className="text-[11px] font-semibold text-foreground leading-none truncate">{provider.displayName}</span>
   377→        <span
   378→          className="inline-block w-1.5 h-1.5 rounded-full shrink-0"
   379→          style={{ backgroundColor: provider.syncedLive ? "#22c55e" : "#f59e0b" }}
   380→          title={provider.syncedLive ? "synced live from provider API" : "config-sourced (no public live API)"}
   381→        />
   382→        <span className="text-[10px] text-muted-foreground leading-none ml-auto tabular-nums">{provider.models.length}</span>
   383→        <button
   384→          onClick={openSettings}
   385→          className="flex items-center justify-center size-5 rounded text-muted-foreground hover:text-foreground hover:bg-muted/60 transition-colors"
   386→          aria-label={`Manage ${provider.displayName} privacy & keys`}
   387→          title={`${provider.manageLabel} \u2197`}
   388→        >
   389→          <IcoSettings />
   390→        </button>
   391→      </div>
   392→
   393→      <button
   394→        onClick={openSettings}
   395→        className="flex items-center gap-1.5 px-2.5 shrink-0 text-left hover:brightness-95 dark:hover:brightness-110 transition-all"
   396→        style={{
   397→          height: 24,
   398→          borderBottom: `1px solid ${confidenceDotColor(provider.privacy.confidence)}25`,
   399→          backgroundColor: `${confidenceDotColor(provider.privacy.confidence)}0d`,
   400→        }}
   401→        title={provider.privacy.notice}
   402→      >
   403→        <span className="shrink-0" style={{ color: confidenceDotColor(provider.privacy.confidence) }}>
   404→          <IcoShield />
   405→        </span>
   406→        <span className="text-[9px] leading-none text-muted-foreground truncate flex-1">
   407→          {provider.privacy.notice}
   408→        </span>
   409→      </button>
   410→
   411→      <div className="flex flex-col gap-px p-1 max-h-[340px] overflow-y-auto">
   412→        {empty ? (
   413→          <div className="flex items-center justify-center h-10 text-[10px] text-muted-foreground/50">
   414→            no models synced
   415→          </div>
   416→        ) : (
   417→          <>
   418→{matched.map((m) => (
   419→              <ModelRow key={m.id} model={m} isSelected={selectedModelId === m.id && (!condensedView ? selectedProviderName === provider.name : true)} onSelect={() => onSelect(m, provider.name)} />
   420→            ))}
   421→            {hasDimmed && (
   422→              <>
   423→                <button
   424→                  onClick={() => setShowDimmed((v) => !v)}
   425→                  className="flex items-center gap-1.5 px-1.5 py-1 mt-0.5 border-t border-border/30 text-[9px] text-muted-foreground/50 hover:text-muted-foreground/80 transition-colors cursor-pointer select-none"
   426→                >
   427→                  <span
   428→                    className="text-[10px] leading-none transition-transform duration-150"
   429→                    style={{ transform: showDimmed ? "rotate(90deg)" : "rotate(0deg)" }}
   430→                  >
   431→                    {"\u25b8"}
   432→                  </span>
   433→                  <span className="text=[9px] leading-none">
   434→                    {showDimmed ? `Hide ${dimmed.length} dimmed` : `Show ${dimmed.length} dimmed`}
   435→                  </span>
   436→                </button>
   437→                {showDimmed && dimmed.map((m) => (
   438→                  <ModelRow key={m.id} model={m} isSelected={selectedModelId === m.id && (!condensedView ? selectedProviderName === provider.name : true)} onSelect={() => onSelect(m, provider.name)} dimmed />
   439→                ))}
   440→              </>
   441→            )}
   442→          </>
   443→        )}
   444→      </div>
   445→    </div>
   446→  );
   447→}

   590→function condensedModelMatchesFilters(model: CondensedModel, activeFilters: string[], contextMin: number): boolean {
   591→  if (contextMin > 0 && model.contextLength < contextMin) return false;
   592→  if (activeFilters.length === 0) return true;
   593→  const caps = model.attributes?.capabilities ?? [];
   594→  const bm = model.attributes?.benchmarks;
   595→  return activeFilters.some((filter) => {
   596→    if (filter === "reasoning") return isReasoningModel(model);
   597→    if (filter === "intelligence") return (bm?.intelligence ?? 0) >= 20;
   598→    if (filter === "code") return caps.includes("code") || (bm?.coding ?? 0) >= 20 || (bm?.aaCoding ?? 0) >= 20;
   599→    if (filter === "agent") return (bm?.agentic ?? 0) >= 20;
   600→    if (filter === "tools") return caps.includes("tools") || caps.includes("tool use") || (bm?.agentic ?? 0) >= 20;
   601→    if (filter === "vision") return caps.includes("vision");
   602→    if (filter === "speech") return caps.includes("speech") || caps.includes("audio");
   603→    return false;
   604→  });
   605→}

   607→function CondensedList({
   608→  models,
   609→  searchQuery,
   610→  activeFilters,
   611→  contextMin,
   612→  hideUnavailable,
   613→  selectedModelId,
   614→  onSelect,
   615→  expandedLogical,
   616→  onToggleExpand,
   617→}: {
   618→  models: CondensedModel[];
   619→  searchQuery: string;
   620→  activeFilters: string[];
   621→  contextMin: number;
   622→  hideUnavailable: boolean;
   623→  selectedModelId: string | null;
   624→  onSelect: (model: CondensedModel) => void;
   625→  expandedLogical: string | null;
   626→  onToggleExpand: (logical: string | null) => void;
   627→}) {
   628→  const [showHidden, setShowHidden] = useState(false);
   629→  const [showUnavailable, setShowUnavailable] = useState(false);
   630→
   631→  const { visible, hiddenByFilter } = useMemo(() => {
   632→    const searched = !searchQuery.trim()
   633→      ? models
   634→      : models.filter((m) => {
   635→          const q = searchQuery.toLowerCase();
   636→          return (
   637→            m.displayName.toLowerCase().includes(q) ||
   638→            m.logical.toLowerCase().includes(q) ||
   639→            (m.family?.toLowerCase().includes(q) ?? false) ||
   640→            m.hosts.some((h) => h.providerDisplayName.toLowerCase().includes(q))
   641→          );
   642→        });
   643→
   644→    const filtMatch: CondensedModel[] = [];
   645→    const filtMiss: CondensedModel[] = [];
   646→    for (const m of searched) {
   647→      if (condensedModelMatchesFilters(m, activeFilters, contextMin)) {
   648→        filtMatch.push(m);
   649→      } else {
   650→        filtMiss.push(m);
   651→      }
   652→    }
   653→
   654→    const sorted = activeFilters.length > 0 || contextMin > 0
   655→      ? [...filtMatch].sort((a, b) => {
   656→          const sa = bestFilterScore({ id: a.logical, displayName: a.displayName, contextLength: a.contextLength, attributes: a.attributes } as ProviderModel, activeFilters);
   657→          const sb = bestFilterScore({ id: b.logical, displayName: b.displayName, contextLength: b.contextLength, attributes: b.attributes } as ProviderModel, activeFilters);
   658→          return sb - sa;
   659→        })
   660→      : defaultSort(filtMatch.map((m) => ({ id: m.logical, displayName: m.displayName, contextLength: m.contextLength, attributes: m.attributes } as ProviderModel))).map((pm) => filtMatch.find((m) => m.logical === pm.id)!).filter(Boolean);
   661→
   662→    return {
   663→      visible: sorted,
   664→      hiddenByFilter: filtMiss,
   665→    };
   666→  }, [models, searchQuery, activeFilters, contextMin]);
   667→
   668→  const selectedLogical = useMemo(() => {
   669→    if (!selectedModelId) return null;
   670→    for (const model of models) {
   671→      if (model.logical === selectedModelId) return model.logical;
   672→      if (model.hosts.some((h) => h.modelId === selectedModelId)) return model.logical;
   673→    }
   674→    return null;
   675→  }, [selectedModelId, models]);
   676→
   677→  const available = useMemo(() => visible.filter((m) => m.hosts.some((h) => h.hasApiKey)), [visible]);
   678→  const unavailable = useMemo(() => visible.filter((m) => !m.hosts.some((h) => h.hasApiKey)), [visible]);
   679→
   680→  return (
   681→    <div className="flex flex-col gap-px p-1">
   682→      {visible.length === 0 && hiddenByFilter.length === 0 && (
   683→        <div className="flex items-center justify-center h-16 text-[10px] text-muted-foreground/50">
   684→          no models match
   685→        </div>
   686→      )}
   687→
   688→      {(hideUnavailable ? available : visible).map((m) => (
   689→        <CondensedModelRow
   690→          key={m.logical}
   691→          model={m}
   692→          isSelected={selectedLogical === m.logical}
   693→          onSelect={() => onSelect(m)}
   694→          expanded={expandedLogical === m.logical}
   695→          onToggleExpand={() => onToggleExpand(expandedLogical === m.logical ? null : m.logical)}
   696→          dimmed={!m.hosts.some((h) => h.hasApiKey)}
   697→        />
   698→      ))}
   699→
   700→      {hideUnavailable && unavailable.length > 0 && (
   701→        <>
   702→          <button
   703→            onClick={() => setShowUnavailable((v) => !v)}
   704→            className="flex items-center gap-1.5 px-1.5 py-1 mt-1 border-t border-border/30 text-[9px] text-muted-foreground/50 hover:text-muted-foreground/80 transition-colors cursor-pointer select-none"
   705→          >
   706→            <span
   707→              className="text-[10px] leading-none transition-transform duration-150"
   708→              style={{ transform: showUnavailable ? "rotate(90deg)" : "rotate(0deg)" }}
   709→            >
   710→              {"\u25b8"}
   711→            </span>
   712→            <span className="text-[9px] leading-none">
   713→              {showUnavailable ? `Hide unavailable (${unavailable.length})` : `Unavailable models (${unavailable.length})`}
   714→            </span>
   715→          </button>
   716→          {showUnavailable && unavailable.map((m) => (
   717→            <CondensedModelRow
   718→              key={m.logical}
   719→              model={m}
   720→              isSelected={selectedLogical === m.logical}
   721→              onSelect={() => onSelect(m)}
   722→              expanded={expandedLogical === m.logical}
   723→              onToggleExpand={() => onToggleExpand(expandedLogical === m.logical ? null : m.logical)}
   724→              dimmed
   725→            />
   726→          ))}
   727→        </>
   728→      )}
   729→
   730→      {hiddenByFilter.length > 0 && (
   731→        <>
   732→          <button
   733→            onClick={() => setShowHidden((v) => !v)}
   734→            className="flex items-center gap-1.5 px-1.5 py-1 mt-1 border-t border-border/30 text-[9px] text-muted-foreground/50 hover:text-muted-foreground/80 transition-colors cursor-pointer select-none"
   735→          >
   736→            <span
   737→              className="text-[10px] leading-none transition-transform duration-150"
   738→              style={{ transform: showHidden ? "rotate(90deg)" : "rotate(0deg)" }}
   739→            >
   740→              {"\u25b8"}
   741→            </span>
   742→            <span className="text-[9px] leading-none">
   743→              {showHidden ? `Hide ${hiddenByFilter.length} hidden` : `Hidden models (${hiddenByFilter.length})`}
   744→            </span>
   745→          </button>
   746→          {showHidden && hiddenByFilter.map((m) => (
   747→            <CondensedModelRow
   748→              key={m.logical}
   749→              model={m}
   750→              isSelected={selectedLogical === m.logical}
   751→              onSelect={() => onSelect(m)}
   752→              expanded={expandedLogical === m.logical}
   753→              onToggleExpand={() => onToggleExpand(expandedLogical === m.logical ? null : m.logical)}
   754→              dimmed
   755→            />
   756→          ))}
   757→        </>
   758→      )}
   759→    </div>
   760→  );
   761→}
