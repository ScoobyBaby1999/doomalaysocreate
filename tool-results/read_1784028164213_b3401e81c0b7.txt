     1→"""SQLite database for GitHub integration: users, workspaces, push logs, registry.
     2→
     3→All state is stored in a single ``doomalaysocreate.db`` file in the service's data directory.
     4→The schema is created/migrated on first access via ``init_db()``.  Thread-safe:
     5→SQLite is accessed with ``check_same_thread=False`` and all writes go through a
     6→serializing ``threading.Lock``.
     7→
     8→stdlib-only at import time; ``cryptography`` (for token encryption) is only
     9→imported when actually encrypting/decrypting tokens.
    10→"""
    11→from __future__ import annotations
    12→
    13→import json
    14→import os
    15→import sqlite3
    16→import threading
    17→import time
    18→import uuid
    19→from pathlib import Path
    20→
    21→DB_PATH = Path(os.environ.get("DOOMALAYSOCREATE_DB_PATH", "/data/doomalaysocreate.db"))
    22→
    23→_write_lock = threading.Lock()
    24→_conn: sqlite3.Connection | None = None
    25→
    26→
    27→def _db() -> sqlite3.Connection:
    28→    global _conn
    29→    if _conn is None:
    30→        _conn = sqlite3.connect(str(DB_PATH), check_same_thread=False)
    31→        _conn.row_factory = sqlite3.Row
    32→        _conn.execute("PRAGMA journal_mode=WAL")
    33→        _conn.execute("PRAGMA foreign_keys=ON")
    34→    return _conn
    35→
    36→
    37→def init_db() -> None:
    38→    """Create tables if they don't exist.  Runs migrations for existing tables."""
    39→    db = _db()
    40→    db.executescript(SCHEMA)
    41→    db.commit()
    42→    # schema migrations for existing databases
    43→    _migrate(db)
    44→
    45→
    46→def _migrate(db: sqlite3.Connection) -> None:
    47→    """Add columns that may not exist in older DBs."""
    48→    for col in (
    49→        "hf_refresh_token_encrypted",
    50→        "hf_token_expires_at",
    51→    ):
    52→        try:
    53→            db.execute(f"ALTER TABLE users ADD COLUMN {col} TEXT")
    54→        except sqlite3.OperationalError:
    55→            pass
    56→    # source_branches is on workspaces, not users
    57→    try:
    58→        db.execute("ALTER TABLE workspaces ADD COLUMN source_branches TEXT")
    59→    except sqlite3.OperationalError:
    60→        pass
    61→    # Phase 3 — last_synced_at on conscious (for stale HF Dataset sync detection)
    62→    try:
    63→        db.execute("ALTER TABLE conscious ADD COLUMN last_synced_at TEXT")
    64→    except sqlite3.OperationalError:
    65→        pass
    66→    # add 'zai' to conscious_agent tier CHECK constraint
    67→    try:
    68→        db.executescript("""
    69→            PRAGMA foreign_keys=OFF;
    70→            CREATE TABLE conscious_agent_v2 (
    71→                id                TEXT PRIMARY KEY,
    72→                conscious_id      TEXT NOT NULL REFERENCES conscious(id) ON DELETE CASCADE,
    73→                role              TEXT NOT NULL,
    74→                model             TEXT NOT NULL,
    75→                tier              TEXT NOT NULL CHECK (tier IN ('claude','open')),
    76→                status            TEXT NOT NULL DEFAULT 'idle' CHECK (status IN ('idle','running','waiting','done','failed')),
    77→                worktree_path     TEXT,
    78→                branch            TEXT,
    79→                parent_agent_id   TEXT,
    80→                subscribed_events TEXT NOT NULL DEFAULT '[]',
    81→                is_orchestrator   INTEGER NOT NULL DEFAULT 0,
    82→                created_at        TEXT NOT NULL DEFAULT (datetime('now')),
    83→                updated_at        TEXT NOT NULL DEFAULT (datetime('now'))
    84→            );
    85→            INSERT INTO conscious_agent_v2 SELECT * FROM conscious_agent;
    86→            DROP TABLE conscious_agent;
    87→            ALTER TABLE conscious_agent_v2 RENAME TO conscious_agent;
    88→            CREATE INDEX IF NOT EXISTS idx_conscious_agent_conscious ON conscious_agent(conscious_id);
    89→            CREATE INDEX IF NOT EXISTS idx_conscious_agent_orchestrator ON conscious_agent(conscious_id, is_orchestrator);
    90→            PRAGMA foreign_keys=ON;
    91→        """)
    92→    except sqlite3.OperationalError:
    93→        pass
    94→    # migrate sandbox_path from /tmp to /data for persist across Space restarts
    95→    try:
    96→        new_base = "/data/workspaces/"
    97→        db.execute(
    98→            "UPDATE workspaces SET sandbox_path = ? || substr(sandbox_path, 16) "
    99→            "WHERE sandbox_path LIKE '/tmp/workspace/%'",
   100→            (new_base,))
   101→        db.execute(
   102→            "UPDATE workspaces SET sandbox_path = ? || substr(sandbox_path, 18) "
   103→            "WHERE sandbox_path LIKE '/tmp/workspaces/%'",
   104→            (new_base,))
   105→    except sqlite3.OperationalError:
   106→        pass
   107→    db.commit()
   108→

   331→def _gen_id() -> str:
   332→    return uuid.uuid4().hex[:16]

   335→def upsert_user(*, user_id: str | None = None, github_id: int | None = None, github_username: str | None = None,
   336→                github_token_encrypted: str | None = None,
   337→                hf_id: str | None = None, hf_username: str | None = None,
   338→                hf_token_encrypted: str | None = None,
   339→                hf_refresh_token_encrypted: str | None = None,
   340→                hf_token_expires_at: str | None = None) -> dict:
   341→    """Create or update a user by user_id (internal UUID), github_id, or hf_id.
   342→    Returns the user row. user_id takes precedence for lookup."""
   343→    db = _db()
   344→    now = _iso_now()
   345→    with _write_lock:
   346→        # find existing
   347→        user = None
   348→        if user_id is not None:
   349→            row = db.execute("SELECT * FROM users WHERE id = ?", (user_id,)).fetchone()
   350→            if row:
   351→                user = dict(row)
   352→        if user is None and github_id is not None:
   353→            row = db.execute("SELECT * FROM users WHERE github_id = ?", (github_id,)).fetchone()
   354→            if row:
   355→                user = dict(row)
   356→        if user is None and hf_id is not None:
   357→            row = db.execute("SELECT * FROM users WHERE hf_id = ?", (hf_id,)).fetchone()
   358→            if row:
   359→                user = dict(row)
   360→        if user is None:
   361→            uid = user_id if user_id else _gen_id()
   362→            db.execute(
   363→                "INSERT INTO users (id, github_id, github_username, github_token_encrypted, "
   364→                "hf_id, hf_username, hf_token_encrypted, "
   365→                "hf_refresh_token_encrypted, hf_token_expires_at, "
   366→                "created_at, updated_at) "
   367→                "VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)",
   368→                (uid, github_id, github_username, github_token_encrypted,
   369→                 hf_id, hf_username, hf_token_encrypted,
   370→                 hf_refresh_token_encrypted, hf_token_expires_at, now, now))
   371→            db.commit()
   372→            return get_user(uid)
   373→        # update
   374→        updates = []
   375→        params = []
   376→        if github_id is not None:
   377→            updates.append("github_id = ?")
   378→            params.append(github_id)
   379→        if github_username is not None:
   380→            updates.append("github_username = ?")
   381→            params.append(github_username)
   382→        if github_token_encrypted is not None:
   383→            updates.append("github_token_encrypted = ?")
   384→            params.append(github_token_encrypted)
   385→        if hf_id is not None:
   386→            updates.append("hf_id = ?")
   387→            params.append(hf_id)
   388→        if hf_username is not None:
   389→            updates.append("hf_username = ?")
   390→            params.append(hf_username)
   391→        if hf_token_encrypted is not None:
   392→            updates.append("hf_token_encrypted = ?")
   393→            params.append(hf_token_encrypted)
   394→        if hf_refresh_token_encrypted is not None:
   395→            updates.append("hf_refresh_token_encrypted = ?")
   396→            params.append(hf_refresh_token_encrypted)
   397→        if hf_token_expires_at is not None:
   398→            updates.append("hf_token_expires_at = ?")
   399→            params.append(hf_token_expires_at)
   400→        if updates:
   401→            updates.append("updated_at = ?")
   402→            params.append(now)
   403→            params.append(user["id"])
   404→            db.execute(f"UPDATE users SET {', '.join(updates)} WHERE id = ?", params)
   405→            db.commit()
   406→    return get_user(user["id"])

   442→def create_workspace(user_id: str, *, title: str, source_repo: str | None = None,
   443→                     source_branch: str | None = None,
   444→                     source_branches: list[str] | None = None,
   445→                     visibility: str = "private",
   446→                     description: str = "", auto_sync: bool = False,
   447→                     sandbox_path: str = "", hf_space_id: str | None = None) -> dict:
   448→    db = _db()
   449→    wid = _gen_id()
   450→    now = _iso_now()
   451→    branches_json = json.dumps(source_branches) if source_branches else None
   452→    with _write_lock:
   453→        db.execute(
   454→            "INSERT INTO workspaces (id, user_id, source_repo, source_branch, "
   455→            "source_branches, sandbox_path, hf_space_id, auto_sync, visibility, "
   456→            "title, description, created_at, last_modified) VALUES "
   457→            "(?,?,?,?,?,?,?,?,?,?,?,?,?)",
   458→            (wid, user_id, source_repo, source_branch, branches_json,
   459→             sandbox_path, hf_space_id,
   460→             int(auto_sync), visibility, title, description, now, now))
   461→        db.commit()
   462→    return get_workspace(wid)

   477→def update_workspace(workspace_id: str, **fields) -> dict | None:
   478→    allowed = {"title", "description", "visibility", "current_branch",
   479→               "hf_space_id", "auto_sync", "last_modified", "sandbox_path",
   480→               "source_branches"}
   481→    updates = []
   482→    params = []
   483→    for k, v in fields.items():
   484→        if k not in allowed:
   485→            continue
   486→        if k == "auto_sync":
   487→            v = int(v)
   488→        elif k == "source_branches" and isinstance(v, list):
   489→            v = json.dumps(v)
   490→        updates.append(f"{k} = ?")
   491→        params.append(v)
   492→    if not updates:
   493→        return get_workspace(workspace_id)
   494→    updates.append("last_modified = ?")
   495→    params.append(_iso_now())
   496→    params.append(workspace_id)
   497→    with _write_lock:
   498→        _db().execute(f"UPDATE workspaces SET {', '.join(updates)} WHERE id = ?", params)
   499→        _db().commit()
   500→    return get_workspace(workspace_id)
   501→

   503→def delete_workspace(workspace_id: str) -> bool:
   504→    with _write_lock:
   505→        db = _db()
   506→        db.execute("DELETE FROM registry WHERE workspace_id = ?", (workspace_id,))
   507→        cur = db.execute("DELETE FROM workspaces WHERE id = ?", (workspace_id,))
   508→        db.commit()
   509→        return cur.rowcount > 0

   572→def list_public_workspaces(*, page: int = 1, per_page: int = 20,
   573→                           sort: str = "recent", search: str = "") -> dict:
   574→    db = _db()
   575→    where = "WHERE 1=1"
   576→    params: list = []
   577→    if search:
   578→        where += " AND (r.title LIKE ? ESCAPE '\\' OR r.description LIKE ? ESCAPE '\\' OR r.owner_username LIKE ? ESCAPE '\\')"
   579→        # escape LIKE wildcards to prevent pattern injection
   580→        escaped = search.replace('%', '\\%').replace('_', '\\_')
   581→        q = "%" + escaped + "%"
   582→        params += [q, q, q]
   583→    order = "r.indexed_at DESC" if sort == "recent" else "r.indexed_count DESC"
   584→    #   indexed_count doesn't exist yet — fall back to indexed_at
   585→    order = "r.indexed_at DESC"
   586→    total = db.execute(f"SELECT COUNT(*) FROM registry r {where}", params).fetchone()[0]
   587→    offset = (max(1, page) - 1) * per_page
   588→    rows = db.execute(
   589→        f"SELECT r.* FROM registry r {where} ORDER BY {order} LIMIT ? OFFSET ?",
   590→        params + [per_page, offset]).fetchall()
   591→    return {"items": [dict(r) for r in rows], "total": total,
   592→            "page": page, "per_page": per_page}

   632→def update_chat_session(session_id: str, **fields) -> dict | None:
   633→    allowed = {"title", "model", "workspace_id"}
   634→    updates = []
   635→    params = []
   636→    for k, v in fields.items():
   637→        if k not in allowed:
   638→            continue
   639→        updates.append(f"{k} = ?")
   640→        params.append(v)
   641→    if not updates:
   642→        return get_chat_session(session_id)
   643→    updates.append("updated_at = ?")
   644→    params.append(_iso_now())
   645→    params.append(session_id)
   646→    with _write_lock:
   647→        _db().execute(f"UPDATE chat_sessions SET {', '.join(updates)} WHERE id = ?", params)
   648→        _db().commit()
   649→    return get_chat_session(session_id)

   660→def append_chat_events(session_id: str, events: list[dict]) -> None:
   661→    """Append events to a chat session, skipping events already stored by seq."""
   662→    if not events:
   663→        return
   664→    db = _db()
   665→    with _write_lock:
   666→        row = db.execute(
   667→            "SELECT COALESCE(MAX(seq), -1) FROM chat_events WHERE session_id = ?",
   668→            (session_id,)).fetchone()
   669→        persisted_max = row[0] if row else -1
   670→        insert_count = 0
   671→        for ev in events:
   672→            i = ev.get("i", -1)
   673→            if i <= persisted_max:
   674→                continue
   675→            db.execute(
   676→                "INSERT INTO chat_events (session_id, seq, event_type, content, created_at) "
   677→                "VALUES (?, ?, ?, ?, ?)",
   678→                (session_id, i, ev.get("type", "unknown"), json.dumps(ev), _iso_now()))
   679→            insert_count += 1
   680→        if insert_count:
   681→            db.commit()
