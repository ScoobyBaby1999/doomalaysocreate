    98→def _build_open_models() -> list[tuple[str, str, str, str | None]]:
    99→    """Build open model entries from providers_catalog.json dynamically.
   100→
   101→    Each configured provider contributes ALL its models (env-var-gated), so
   102→    every model from NVIDIA, Cloudflare, PrivateMode AI etc. is available in
   103→    the agent without hardcoding model names.
   104→    """
   105→    global _open_models_cache
   106→    # Re-probe if the cache is empty — the panel sync may not have completed
   107→    # on the first call (cache poisoning fix). A non-empty cache is still
   108→    # trusted (the catalog + synced models don't change at runtime).
   109→    if _open_models_cache is not None and len(_open_models_cache) > 0:
   110→        return _open_models_cache
   111→
   112→    entries: list[tuple[str, str, str, str | None]] = []
   113→    catalog_path = HERE / "providers_catalog.json"
   114→    try:
   115→        with open(catalog_path, encoding="utf-8") as f:
   116→            data = json.load(f)
   117→    except (OSError, json.JSONDecodeError):
   118→        _open_models_cache = entries
   119→        return entries
   120→
   121→    for prov in data.get("providers", []):
   122→        name = prov["name"]
   123→        env_var = prov.get("env_var", "")
   124→        if isinstance(env_var, list):
   125→            env_var = env_var[0] if env_var else ""
   126→        if not env_var:
   127→            continue
   128→
   129→        # Resolve {VAR} placeholders in base_url from the environment.
   130→        base_url = prov.get("base_url", "")
   131→        for var in prov.get("requires", []):
   132→            base_url = base_url.replace("{" + var + "}", os.environ.get(var, "").strip())
   133→        # LiteLLM appends /chat/completions for openai/ models, so strip it.
   134→        if base_url.endswith("/chat/completions"):
   135→            base_url = base_url[:-len("/chat/completions")]
   136→
   137→        for model_id in prov.get("models", []):
   138→            litellm_model = f"openai/{model_id}"
   139→            label = f"{model_id} ({prov.get('displayName', name)})"
   140→            entries.append((env_var, label, litellm_model, base_url or None))
   141→
   142→    # Also include dynamically synced models from providers with sync_config.
   143→    # The panel syncs models from live /v1/models endpoints at boot; those are
   144→    # cached in provider_sync._panel_sync_cache.  Providers whose env var is set
   145→    # contribute ALL their synced models so the agent picker shows everything
   146→    # the panel can route to, not just the static catalog.
   147→    from provider_sync import get_panel_sync_cache
   148→    sync_cache = get_panel_sync_cache()
   149→    if sync_cache:
   150→        seen: set[str] = set(m.split("/")[-1] for _, _, m, _ in entries)
   151→        for prov in data.get("providers", []):
   152→            name = prov["name"]
   153→            sync_config = prov.get("sync_config")
   154→            if not sync_config or not sync_config.get("enabled"):
   155→                continue
   156→            senv = prov.get("env_var", "")
   157→            if isinstance(senv, list):
   158→                senv = senv[0] if senv else ""
   159→            if not senv or not os.environ.get(senv, "").strip():
   160→                continue
   161→            synced = sync_cache.get(name, [])
   162→            if not synced:
   163→                continue
   164→            sbase = prov.get("base_url", "")
   165→            for var in prov.get("requires", []):
   166→                sbase = sbase.replace("{" + var + "}", os.environ.get(var, "").strip())
   167→            if sbase.endswith("/chat/completions"):
   168→                sbase = sbase[:-len("/chat/completions")]
   169→            for mid in synced:
   170→                last = mid.split("/")[-1]
   171→                if last not in seen:
   172→                    seen.add(last)
   173→                    entries.append((senv, f"{mid} ({prov.get('displayName', name)})",
   174→                                    f"openai/{mid}", sbase or None))
   175→
   176→    _open_models_cache = entries
   177→    return entries

   180→def _resolve_open_model(user_model: str) -> tuple[str, str | None, str | None, str | None] | None:
   181→    """Match a user-provided model name to a litellm model string + base_url +
   182→    env_var + provider name from the dynamic open-models list.
   183→
   184→    Handles all input formats:
   185→    - Canonical litellm: ``openai/kimi-k2.6`` (exact match)
   186→    - Panel provider/model: ``privatemodeai/kimi-k2.6`` (match by last segment)
   187→    - Logical/bare name: ``kimi-k2.6`` (match by last segment)
   188→    - Partial: ``deepseek-v4-flash`` (match by last segment)
   189→
   190→    Prefers an exact full match, then falls back to last-segment matching.
   191→    When the same model name exists on multiple providers, the first one in
   192→    catalog order wins (same as _build_open_models dedup). To route to a
   193→    specific provider, pass ``provider/model`` — the provider prefix is
   194→    matched against the env_var's provider name.
   195→
   196→    Returns ``(litellm_model, base_url, env_var, provider_name)`` or ``None``.
   197→    """
   198→    if not user_model:
   199→        return None
   200→    user_last = user_model.split("/")[-1]
   201→    # Also extract a provider hint if the user passed "provider/model"
   202→    user_provider = user_model.split("/")[0] if "/" in user_model else ""
   203→    models = _build_open_models()
   204→    # Pass 1: exact full match (canonical litellm string)
   205→    for _env, _label, model, base in models:
   206→        if model == user_model:
   207→            return (model, base, _env, _label)
   208→    # Pass 2: match by provider hint + last segment (e.g. "privatemodeai/kimi-k2.6")
   209→    if user_provider:
   210→        for env, label, model, base in models:
   211→            model_last = model.split("/")[-1]
   212→            # label looks like "kimi-k2.6 (PrivateMode AI)" — extract provider
   213→            label_provider = ""
   214→            if "(" in label and ")" in label:
   215→                label_provider = label[ label.index("(") + 1 : label.rindex(")") ].strip().lower()
   216→            # env var maps to provider: PRIVATEMODEAI_API_KEY -> privatemodeai
   217→            env_provider = env.lower().replace("_api_key", "").replace("_token", "").replace("_api_token", "")
   218→            if (model_last == user_last and
   219→                (user_provider.lower() in env_provider or
   220→                 user_provider.lower() in label_provider.lower())):
   221→                return (model, base, env, label)
   222→    # Pass 3: last-segment match (bare logical names, partial names)
   223→    for env, label, model, base in models:
   224→        if model.split("/")[-1] == user_last:
   225→            return (model, base, env, label)
   226→    return None

   304→def agent_tier() -> str | None:
   305→    """Which agent tier this Space can actually run: "claude" | "open" | "mock".
   306→
   307→    Requires BOTH a key and the matching SDK installed for a real tier — so
   308→    /health never advertises a tier the worker can't start.  Falls back to
   309→    "mock" when no real tier is available so duplicated Spaces (which start
   310→    with zero provider keys) still let users dogfood the agent UI instead of
   311→    hitting a 503.  Once a user adds a provider key via the onboarding wizard,
   312→    HF restarts the Space and the real tier takes over automatically.
   313→    Set AGENT_FORCE_TIER=none to disable the mock fallback (old behaviour).
   314→    """
   315→    forced = os.environ.get("AGENT_FORCE_TIER", "").strip().lower()
   316→    if forced in ("claude", "open", "mock"):
   317→        return forced
   318→    if os.environ.get("ANTHROPIC_API_KEY", "").strip() and _installed("claude_agent_sdk"):
   319→        return "claude"
   320→    if _pick_open_llm() is not None and _open_sdk_installed():
   321→        return "open"
   322→    # Auto-fallback: mock tier so the agent UI is always usable on fresh
   323→    # duplicated Spaces that have no provider keys yet.
   324→    if forced != "none":
   325→        return "mock"
   326→    return None

   521→class ClaudeAdapter(BaseAdapter):

   562→        return getattr(self, "_session", None)

   564→    def turn(self, user_msg: str, emit) -> None:
   565→        assert self.loop is not None
   566→        self.loop.run_until_complete(self._turn(user_msg, emit))
   567→
   568→    def interrupt(self) -> None:
   569→        #   the loop is owned by the worker thread; schedule the SDK's interrupt
   570→        #   coroutine onto it from the caller's (HTTP handler) thread.
   571→        if self.loop is None or self.client is None or self.loop.is_closed():
   572→            return
   573→        try:
   574→            asyncio.run_coroutine_threadsafe(self.client.interrupt(), self.loop)
   575→        except Exception:
   576→            pass

   578→    async def _turn(self, user_msg: str, emit) -> None:
   579→        sdk = self._sdk
   580→        thinking_cls = getattr(sdk, "ThinkingBlock", None)
   581→        await self.client.query(user_msg)
   582→        async for msg in self.client.receive_response():
   583→            # Phase 2 (completed in Phase 5 pass) — mid-turn cost abort.
   584→            # After each message received, check the conscious cost ceiling.
   585→            # If over, interrupt the SDK and emit a cost.exceeded event so the
   586→            # orchestrator's next conscious_context surfaces it. This is the
   587→            # "hard enforcement mid-turn" from TIER3_PLAN §14 (the deeper SDK
   588→            # hook that the turn-boundary check was the baseline for).
   589→            sess = getattr(self, "_session", None)
   590→            if sess is not None and getattr(sess, "conscious_id", None):
   591→                if not _cost_turn_ok(sess.conscious_id):
   592→                    spent, ceiling = _cost_figures(sess.conscious_id)
   593→                    emit({"type": "status", "state": "idle",
   594→                          "detail": f"cost ceiling exceeded mid-turn (spent=${spent:.4f}, ceiling=${ceiling:.4f})"})
   595→                    try:
   596→                        await self.client.interrupt()
   597→                    except Exception:
   598→                        pass
   599→                    try:
   600→                        import conscious_db
   601→                        conscious_db.append_event(
   602→                            sess.conscious_id, "cost.exceeded",
   603→                            f"mid-turn abort: spent=${spent:.4f} ceiling=${ceiling:.4f}",
   604→                            author=getattr(sess, "agent_id", None))
   605→                    except Exception:
   606→                        pass
   607→                    return
   608→            if isinstance(msg, sdk.AssistantMessage):
   609→                for block in msg.content:
   610→                    if isinstance(block, sdk.TextBlock):
   611→                        emit({"type": "assistant", "text": block.text})
   612→                    elif thinking_cls is not None and isinstance(block, thinking_cls):
   613→                        emit({"type": "thinking", "text": _clip(block.thinking)})
   614→                    elif isinstance(block, sdk.ToolUseBlock):
   615→                        emit({"type": "tool_use", "name": block.name,
   616→                              "summary": _summarize_tool_input(block.name, block.input)})
   617→            elif isinstance(msg, sdk.UserMessage):
   618→                content = msg.content if isinstance(msg.content, list) else []
   619→                for block in content:
   620→                    if isinstance(block, sdk.ToolResultBlock):
   621→                        emit({"type": "tool_result", "text": _clip(block.content),
   622→                              "is_error": bool(getattr(block, "is_error", False))})
   623→            elif isinstance(msg, sdk.ResultMessage):
   624→                emit({"type": "status", "state": "idle",
   625→                      "usage": getattr(msg, "usage", None),
   626→                      "cost_usd": getattr(msg, "total_cost_usd", None)})
   627→
   628→    def close(self) -> None:
   629→        if self.loop is None:
   630→            return
   631→        try:
   632→            if self.client is not None:
   633→                self.loop.run_until_complete(self.client.disconnect())
   634→        except Exception:
   635→            pass
   636→        finally:
   637→            self.loop.close()

   640→def _parse_git_branch(cmd: str) -> str | None:
   641→    """Best-effort extract a branch name from a git push/pull/fetch command.
   642→    Returns None if no explicit branch is named (caller should use the
   643→    current branch via ``_current_branch``)."""
   644→    parts = cmd.split()
   645→    # skip 'git' and the subcommand; ignore flags and the 'origin' keyword
   646→    candidates = [p for p in parts[2:] if p and not p.startswith("-") and p != "origin"]
   647→    return candidates[0] if candidates else None
   648→
   649→
   650→def _current_branch(sandbox: str) -> str:
   651→    """Return the current checked-out branch name, or 'main' as a fallback."""
   652→    import subprocess
   653→    r = subprocess.run(["git", "-C", sandbox, "rev-parse", "--abbrev-ref", "HEAD"],
   654→                       capture_output=True, text=True, timeout=10)
   655→    return r.stdout.strip() or "main"

   658→def _route_network_git(cmd: str, workspace_id: str) -> dict:
   659→    """Route agent git push/pull/fetch through the backend git functions so
   660→    the GitHub token is never written to ``.git/config`` and every push is
   661→    audit-logged.  Returns a Strands-style tool result dict.
   662→    """
   663→    import db
   664→    import github_integration
   665→    ws = db.get_workspace(workspace_id)
   666→    if not ws:
   667→        return {"status": "error",
   668→                "content": [{"text": f"workspace not found: {workspace_id}"}]}
   669→    user_id = ws["user_id"]
   670→    sandbox = ws["sandbox_path"]
   671→    stripped = cmd.strip()
   672→    try:
   673→        if stripped.startswith("git push"):
   674→            force = ("-f " in stripped) or ("--force" in stripped)
   675→            branch = _parse_git_branch(stripped) or _current_branch(sandbox)
   676→            github_integration.push_to_remote(
   677→                user_id, workspace_id, branch, force=force)
   678→            return {"status": "success",
   679→                    "content": [{"text": f"pushed {branch} to origin"}]}
   680→        if stripped.startswith("git fetch"):
   681→            branch = _parse_git_branch(stripped) or ""
   682→            github_integration.fetch_from_remote(user_id, workspace_id, branch)
   683→            return {"status": "success",
   684→                    "content": [{"text": "fetched from origin" +
   685→                                (f" ({branch})" if branch else "")}]}
   686→        if stripped.startswith("git pull"):
   687→            branch = _parse_git_branch(stripped) or _current_branch(sandbox)
   688→            github_integration.fetch_from_remote(user_id, workspace_id, branch)
   689→            return {"status": "success",
   690→                    "content": [{"text": f"pulled {branch} from origin"}]}
   691→    except Exception as exc:  # noqa: BLE001 - surface git errors to the agent
   692→        return {"status": "error",
   693→                "content": [{"text": f"git operation failed: {exc}"}]}
   694→    return {"status": "error",
   695→            "content": [{"text": f"unsupported git command: {cmd}"}]}
   696→

   698→def _guarded_shell(**kwargs):
   699→    """Wrap the Strands shell tool: force workspace dir + intercept git commands.
   700→
   701→    1. Overrides ``workdir`` with the session workspace from thread-local storage.
   702→    2. Network git operations (push/pull/fetch) are routed through the backend
   703→       git functions (``push_to_remote`` / ``fetch_from_remote``) so the GitHub
   704→       token never lands in ``.git/config`` and pushes are audit-logged.
   705→    """
   706→    from strands_tools import shell as _shell
   707→
   708→    kwargs["workdir"] = str(getattr(_thread_local, "workspace", Path.cwd()))
   709→
   710→    cmd = kwargs.get("command", "")
   711→    if isinstance(cmd, str) and cmd.strip():
   712→        stripped = cmd.strip()
   713→        workspace_id = getattr(_thread_local, "workspace_id", None)
   714→        if workspace_id and (
   715→            stripped.startswith("git push")
   716→            or stripped.startswith("git pull")
   717→            or stripped.startswith("git fetch")
   718→        ):
   719→            is_force = stripped.startswith("git push") and (
   720→                "-f " in stripped or "--force" in stripped)
   721→            if not is_force:
   722→                return _route_network_git(stripped, workspace_id)
   723→    return _shell.tool(**kwargs)

   856→    def turn(self, user_msg: str, emit) -> None:
   857→        _thread_local.workspace = self.workspace
   858→        _thread_local.workspace_id = self.workspace_id
   859→        sess = getattr(self, "_session", None)
   860→        # Track whether thinking was emitted via the streaming callback.
   861→        # If so, skip thinking emission in the post-turn walk (which would
   862→        # duplicate the text). Reset at the start of each turn.
   863→        self._thinking_streamed = False
   864→        # Build a streaming callback handler: thinking text in real-time,
   865→        # plus mid-turn cost ceiling enforcement for conscious agents.
   866→        def _stream_callback(**kw):
   867→            reasoning = kw.get("reasoningText")
   868→            if reasoning:
   869→                self._thinking_streamed = True
   870→                emit({"type": "thinking", "text": reasoning})
   871→            # Cost ceiling check (conscious agents only)
   872→            if sess is not None and getattr(sess, "conscious_id", None):
   873→                if not _cost_turn_ok(sess.conscious_id):
   874→                    spent, ceiling = _cost_figures(sess.conscious_id)
   875→                    emit({"type": "status", "state": "idle",
   876→                          "detail": f"cost ceiling exceeded mid-turn (spent=${spent:.4f}, ceiling=${ceiling:.4f})"})
   877→                    canceler = getattr(self.agent, "cancel", None)
   878→                    if callable(canceler):
   879→                        try: canceler()
   880→                        except Exception: pass
   881→                    try:
   882→                        import conscious_db
   883→                        conscious_db.append_event(
   884→                            sess.conscious_id, "cost.exceeded",
   885→                            f"mid-turn abort: spent=${spent:.4f} ceiling=${ceiling:.4f}",
   886→                            author=getattr(sess, "agent_id", None))
   887→                    except Exception: pass
   888→        try:
   889→            self.agent.callback_handler = _stream_callback
   890→        except Exception:
   891→            pass
   892→        try:
   893→            self.agent(user_msg)
   894→        finally:
   895→            pass
   896→        # Walk newly-appended messages for tool results and final assistant text
   897→        msgs = getattr(self.agent, "messages", []) or []
   898→        for m in msgs[self._msg_cursor:]:
   899→            role = m.get("role")
   900→            for block in (m.get("content") or []):
   901→                if "toolUse" in block:
   902→                    tu = block["toolUse"] or {}
   903→                    emit({"type": "tool_use", "name": tu.get("name", "tool"),
   904→                          "summary": _summarize_tool_input(tu.get("name", ""),
   905→                                                            tu.get("input", {}))})
   906→                elif "toolResult" in block:
   907→                    tr = block["toolResult"] or {}
   908→                    parts = []
   909→                    for c in (tr.get("content") or []):
   910→                        if isinstance(c, dict) and "text" in c:
   911→                            parts.append(c["text"])
   912→                    emit({"type": "tool_result", "text": _clip("\n".join(parts) or tr),
   913→                          "is_error": tr.get("status") == "error"})
   914→                elif "reasoningContent" in block:
   915→                    # Skip if the streaming callback already captured thinking.
   916→                    # The streaming callback emits fragments in real-time which
   917→                    # emit() accumulates into one event. Re-emitting the full
   918→                    # text here would create a duplicate thinking bubble.
   919→                    if getattr(self, "_thinking_streamed", False):
   920→                        continue
   921→                    rc = block["reasoningContent"] or {}
   922→                    txt = (rc.get("reasoningText") or {}).get("text") if isinstance(
   923→                        rc.get("reasoningText"), dict) else rc.get("reasoningText")
   924→                    if txt:
   925→                        emit({"type": "thinking", "text": _clip(txt)})
   926→                elif "text" in block and role == "assistant":
   927→                    if block["text"].strip():
   928→                        emit({"type": "assistant", "text": block["text"]})
   929→        self._msg_cursor = len(msgs)
   930→
   931→    def interrupt(self) -> None:
   932→        canceler = getattr(self.agent, "cancel", None)
   933→        if callable(canceler):
   934→            try:
   935→                canceler()
   936→            except Exception:
   937→                pass

   941→        if callable(cleanup):
   942→            try:
   943→                cleanup()
   944→            except Exception:
   945→                pass

  1067→    # -- transcript ---------------------------------------------------------
  1068→    def emit(self, ev: dict) -> None:
  1069→        with self.lock:
  1070→            if (ev.get("type") == "thinking"
  1071→                    and self.events and self.events[-1].get("type") == "thinking"):
  1072→                # Always APPEND to the last thinking event. The Strands callback
  1073→                # sends reasoning fragments (one per token or chunk). Tool calls
  1074→                # (tool_use/tool_result) naturally separate reasoning blocks —
  1075→                # after a tool_result, self.events[-1] is tool_result (not
  1076→                # thinking), so a new thinking event is created automatically.
  1077→                # This means consecutive thinking events are ALWAYS continuations
  1078→                # of the same reasoning block, so appending is correct.
  1079→                old_text = (self.events[-1].get("text", "") or "")
  1080→                new_text = (ev.get("text", "") or "")
  1081→                # Skip stale duplicates (new text is a prefix of old)
  1082→                if old_text and new_text and old_text.startswith(new_text) and len(new_text) < len(old_text):
  1083→                    stream_ev = {"i": self.events[-1]["i"], "ts": self.events[-1]["ts"], **ev}
  1084→                    stream_ev["text"] = old_text
  1085→                else:
  1086→                    self.events[-1]["text"] = old_text + new_text
  1087→                    self.events[-1]["ts"] = time.time()
  1088→                    stream_ev = {"i": self.events[-1]["i"], "ts": self.events[-1]["ts"], **ev}
  1089→                    stream_ev["text"] = self.events[-1]["text"]
  1090→            else:
  1091→                new_ev = {"i": len(self.events), "ts": time.time(), **ev}
  1092→                self.events.append(new_ev)
  1093→                stream_ev = new_ev
  1094→            self.updated = time.time()
  1095→            queues = list(self._stream_queues)
  1096→        for q in queues:
  1097→            try:
  1098→                q.put_nowait(stream_ev)
  1099→            except (queue.Full, ValueError):
  1100→                try:
  1101→                    self._stream_queues.remove(q)
  1102→                except ValueError:
  1103→                    pass
  1104→
  1105→    def register_stream_queue(self, q: queue.Queue) -> None:
  1106→        with self.lock:
  1107→            self._stream_queues.append(q)
  1108→
  1109→    def unregister_stream_queue(self, q: queue.Queue) -> None:
  1110→        with self.lock:
  1111→            try:
  1112→                self._stream_queues.remove(q)
  1113→            except ValueError:
  1114→                pass

  1116→    def subscribe(self, since: int = 0):
  1117→        """Subscribe to live events for SSE streaming. Returns a queue.Queue
  1118→        that receives every new event as it's emitted. Events with i < since
  1119→        are replayed first (so a late subscriber catches up), then the queue
  1120→        stays open for live events. Call unsubscribe(q) when done.
  1121→
  1122→        This is the method the SSE endpoint looks for via getattr(session,
  1123→        'subscribe', None). Without it, the SSE handler falls back to polling
  1124→        (200ms snapshots) — which is why streaming felt choppy/non-live."""
  1125→        q: queue.Queue = queue.Queue(maxsize=256)
  1126→        # Replay events the subscriber hasn't seen yet.
  1127→        with self.lock:
  1128→            replay = [e for e in self.events if (e.get("i", 0) >= since)]
  1129→        for ev in replay:
  1130→            try:
  1131→                q.put_nowait(ev)
  1132→            except queue.Full:
  1133→                pass  # drop backlog if subscriber is slow
  1134→        self.register_stream_queue(q)
  1135→        return q
  1136→
  1137→    def unsubscribe(self, q: queue.Queue) -> None:
  1138→        """Unsubscribe from live events. Pushes None to signal end-of-stream."""
  1139→        self.unregister_stream_queue(q)
  1140→        try:
  1141→            q.put_nowait(None)
  1142→        except queue.Full:
  1143→            pass

  1183→    def _run(self) -> None:
  1184→        adapter = _make_adapter(self.tier, self.workspace, self.model,
  1185→                                self.workspace_id, self.system_prompt)
  1186→        self.adapter = adapter
  1187→        # Tier 3 — give the adapter a back-reference so it can register the
  1188→        # conscious tool registry on the owning session (conscious_id/agent_id).
  1189→        adapter._session = self
  1190→        try:
  1191→            adapter.open()
  1192→        except Exception as exc:
  1193→            self._set_status("error", detail=f"agent init failed: {_clip(str(exc), 300)}")
  1194→            return
  1195→        self._set_status("idle")
  1196→        while not self.closed:
  1197→            try:
  1198→                msg = self.inbox.get(timeout=30)
  1199→            except queue.Empty:
  1200→                if time.time() - self.updated > SESSION_TTL_S:
  1201→                    break
  1202→                continue
  1203→            if msg is None:
  1204→                break
  1205→            self._interrupting = False
  1206→            self._set_status("running")
  1207→            self.emit({"type": "user", "text": msg})
  1208→            # Phase 2 — cost-ceiling turn-boundary check (§14 hard enforcement).
  1209→            # If over ceiling before the turn even starts, refuse to run and
  1210→            # emit a cost.exceeded event. Mid-turn abort needs deeper SDK hooks
  1211→            # (callback handler inspecting each tool result); the turn-boundary
  1212→            # check is the safe baseline that works in both adapters.
  1213→            if self.conscious_id and not _cost_turn_ok(self.conscious_id):
  1214→                spent, ceiling = _cost_figures(self.conscious_id)
  1215→                self.emit({"type": "status", "state": "idle",
  1216→                           "detail": f"cost ceiling exceeded (spent=${spent:.4f}, ceiling=${ceiling:.4f})"})
  1217→                try:
  1218→                    import conscious_db
  1219→                    conscious_db.append_event(
  1220→                        self.conscious_id, "cost.exceeded",
  1221→                        f"turn refused: spent=${spent:.4f} ceiling=${ceiling:.4f}",
  1222→                        author=self.agent_id)
  1223→                except Exception:
  1224→                    pass
  1225→                self._set_status("idle", detail="cost ceiling exceeded")
  1226→                continue
  1227→            try:
  1228→                adapter.turn(msg, self.emit)
  1229→                if self._interrupting:
  1230→                    self._set_status("idle", detail="interrupted")
  1231→                else:
  1232→                    self._set_status("idle")
  1233→            except Exception as exc:
  1234→                if self._interrupting:
  1235→                    self._set_status("idle", detail="interrupted")
  1236→                else:
  1237→                    self._set_status("error", detail=_clip(str(exc), 300))
  1238→            finally:
  1239→                self._interrupting = False
  1240→        adapter.close()

  1247→def _sweep_locked() -> None:
  1248→    now = time.time()
  1249→    expired = [sid for sid, s in _sessions.items()
  1250→               if now - s.updated > SESSION_TTL_S or s.closed]
  1251→    for sid in expired:
  1252→        s = _sessions.pop(sid)
  1253→        s.close()
  1254→        shutil.rmtree(s.workspace, ignore_errors=True)

  1258→    with _sessions_lock:
  1259→        _sweep_locked()
  1260→        return _sessions.get(session_id)

  1263→def _normalize_model_for_compare(model: str | None) -> str | None:
  1264→    """Normalize a model string for comparison so that equivalent models from
  1265→    different formats compare equal. E.g. all of these -> "kimi-k2.6":
  1266→      "kimi-k2.6", "openai/kimi-k2.6", "privatemodeai/kimi-k2.6"
  1267→    Returns None if model is None/empty.
  1268→    """
  1269→    if not model:
  1270→        return None
  1271→    m = model.strip()
  1272→    if not m:
  1273→        return None
  1274→    # Strip the litellm "openai/" prefix and any "provider/" prefix
  1275→    parts = m.split("/")
  1276→    return parts[-1].lower()

  1279→def get_or_create(session_id: str | None = None,
  1280→                  model: str | None = None,
  1281→                  workspace_id: str | None = None,
  1282→                  conscious_id: str | None = None,
  1283→                  agent_id: str | None = None,
  1284→                  chat_session_id: str | None = None) -> AgentSession:
  1285→    """Reuse a live session by id, or start a new one (CapacityError if full).
  1286→    `model` (optional) selects which model/tier drives a NEW session.
  1287→    `workspace_id` (optional) links the session to a user workspace sandbox.
  1288→    `conscious_id` + `agent_id` (optional, Tier 3) bind the session to a
  1289→    Conscious agent row so the conscious_* tools resolve context.
  1290→    `chat_session_id` (optional) links this agent session to a persistent
  1291→    chat session for event persistence.
  1292→    """
  1293→    tier = tier_for_model(model)
  1294→    if tier is None:
  1295→        raise RuntimeError("no agent tier available for the requested model")
  1296→    with _sessions_lock:
  1297→        _sweep_locked()
  1298→        if session_id and session_id in _sessions:
  1299→            return _sessions[session_id]
  1300→        # Reuse existing agent session linked to this chat_session_id — BUT
  1301→        # only if the model hasn't changed. If the user switched models mid-
  1302→        # conversation, we must create a fresh agent session so the new model
  1303→        # actually takes effect (the old session's adapter is pinned to the
  1304→        # old model). The old session is closed to free its resources.
  1305→        if chat_session_id:
  1306→            existing = None
  1307→            for s in _sessions.values():
  1308→                if s.chat_session_id == chat_session_id:
  1309→                    existing = s
  1310→                    break
  1311→            if existing is not None:
  1312→                # Normalize both model strings for comparison so
  1313→                # "kimi-k2.6" and "openai/kimi-k2.6" are treated as the same.
  1314→                norm_existing = _normalize_model_for_compare(existing.model)
  1315→                norm_new = _normalize_model_for_compare(model)
  1316→                if norm_new is None or norm_existing == norm_new:
  1317→                    return existing
  1318→                # Model changed — close the old session and fall through to
  1319→                # create a new one with the new model.
  1320→                existing.close()
  1321→                _sessions.pop(existing.id, None)
  1322→        if len(_sessions) >= MAX_SESSIONS:
  1323→            raise CapacityError(f"max {MAX_SESSIONS} concurrent agent sessions")
  1324→        # resolve workspace_id to a filesystem path
  1325→        workspace_path = None
  1326→        if workspace_id:
  1327→            import db
  1328→            import github_integration
  1329→            # repair sandbox if missing (Space restart wiped /data/)
  1330→            github_integration.ensure_workspace_sandbox(workspace_id)
  1331→            ws = db.get_workspace(workspace_id)
  1332→            if ws and ws.get("sandbox_path"):
  1333→                workspace_path = Path(ws["sandbox_path"])
  1334→        s = AgentSession(tier, model, workspace_path=workspace_path,
  1335→                         workspace_id=workspace_id,
  1336→                         conscious_id=conscious_id, agent_id=agent_id,
  1337→                         chat_session_id=chat_session_id)
  1338→        _sessions[s.id] = s
  1339→        return s
  1340→

  1346→def _cost_figures(conscious_id: str) -> tuple[float, float]:
  1347→    """Return (spent, ceiling) for a conscious. ceiling=0 means infinite."""
  1348→    try:
  1349→        import conscious_db
  1350→        c = conscious_db.get_conscious(conscious_id) or {}
  1351→        return (float(c.get("cost_spent_usd", 0) or 0),
  1352→                float(c.get("cost_ceiling_usd", 0) or 0))
  1353→    except Exception:
  1354→        return (0.0, 0.0)

  1357→def _cost_turn_ok(conscious_id: str) -> bool:
  1358→    """True if the conscious is under its cost ceiling (or ceiling=0=infinite)."""
  1359→    spent, ceiling = _cost_figures(conscious_id)
  1360→    if ceiling <= 0:
  1361→        return True
  1362→    return spent < ceiling
